+"""Docker runtimeClass adapter.
+
+The tool process is created inside the container. runsc is used when the
+daemon lists it; otherwise runc is the isolation capability. A requested
+runtime that is not installed fails closed. Nothing here starts the tool as
+a server-host subprocess.
+"""
+
+from __future__ import annotations
+
+import json
+import os
+import re
+import shutil
+import subprocess
+import uuid
+from pathlib import Path
+
+from rxycode_runner.sandbox.provider import (
+    AttachRefused,
+    ExecResult,
+    IsolationUnavailable,
+    SandboxHandle,
+)
+
+_ID = re.compile(r"^[A-Za-z0-9._:-]{1,64}$")
+_SANDBOX = re.compile(r"^rxy-sbx-[0-9a-f]{12}$")
+_IMAGE = re.compile(r"^[A-Za-z0-9._:/-]{1,128}$")
+_HOSTNAME = re.compile(r"^[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?$")
+_RUNTIME = re.compile(r"^[A-Za-z0-9._:-]{1,64}$")

+def docker_executable() -> str:
+    """Return the docker CLI. A missing CLI is an unavailable target."""
+    override = os.environ.get("RXYCODE_DOCKER", "").strip()
+    if override:
+        if Path(override).is_file():
+            return override
+        raise IsolationUnavailable("docker", "RXYCODE_DOCKER is not a docker executable")
+    found = shutil.which("docker")
+    if found:
+        return found
+    program_files = os.environ.get("ProgramFiles", r"C:\Program Files")
+    candidates = (
+        Path(program_files) / "Docker" / "Docker" / "resources" / "bin" / "docker.exe",
+        Path(r"E:\DockerDesktop\resources\bin\docker.exe"),
+    )
+    for candidate in candidates:
+        if candidate.is_file():
+            return str(candidate)
+    raise IsolationUnavailable("docker", "docker executable is not installed")
+
+
+class GVisorSandbox:
+    """runc/runsc runtimeClass backend. Labels are the tenant source of truth."""
+
+    def __init__(
+        self,
+        *,
+        runtime: str | None = None,
+        image: str = "python:3.12-slim",
+        hostname: str = "rxy-isolated",
+        docker_bin: str | None = None,
+    ) -> None:
+        if not _IMAGE.fullmatch(image):
+            raise IsolationUnavailable(image, "image name is refused")
+        if not _HOSTNAME.fullmatch(hostname):
+            raise IsolationUnavailable(hostname, "hostname is refused")
+        if runtime is not None and not _RUNTIME.fullmatch(runtime):
+            raise IsolationUnavailable("runtime", "runtime class name is refused")
+        self._runtime = runtime
+        self._image = image
+        self._hostname = hostname
+        self._docker_bin = docker_bin or docker_executable()
+        if not Path(self._docker_bin).is_file():
+            raise IsolationUnavailable("docker", "docker executable is not installed")
+
+    def installed_runtimes(self) -> list[str]:
+        proc = self._docker(["info", "--format", "{{json .Runtimes}}"], timeout=30)
+        if proc.returncode != 0 or not proc.stdout.strip():
+            detail = (proc.stderr or proc.stdout or "docker info failed").strip()
+            raise IsolationUnavailable("docker", detail)
+        try:
+            payload = json.loads(proc.stdout)
+        except json.JSONDecodeError as exc:
+            raise IsolationUnavailable("docker", "docker info returned no runtime map") from exc
+        if not isinstance(payload, dict):
+            raise IsolationUnavailable("docker", "docker info returned no runtime map")
+        return sorted(str(name) for name in payload)
+
+    def create(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        retain: bool = True,
+    ) -> SandboxHandle:
+        _require_id(tenant_id)
+        _require_id(workspace_id)
+        runtime = self._select_runtime()
+        sandbox_id = "rxy-sbx-" + uuid.uuid4().hex[:12]
+        created = self._docker(
+            [
+                "create",
+                "--name",
+                sandbox_id,
+                "--hostname",
+                self._hostname,
+                "--network",
+                "none",
+                "--cap-drop",
+                "ALL",
+                "--security-opt",
+                "no-new-privileges",
+                "--memory",
+                "256m",
+                "--pids-limit",
+                "64",
+                "--runtime",
+                runtime,
+                "--label",

+                "--label",
+                f"rxy.tenant={tenant_id}",
+                "--label",
+                f"rxy.workspace={workspace_id}",
+                "--label",
+                f"rxy.sandbox={sandbox_id}",
+                "--label",
+                f"rxy.retain={'true' if retain else 'false'}",
+                self._image,
+                "sleep",
+                "infinity",
+            ],
+            timeout=60,
+        )
+        if created.returncode != 0:
+            detail = (created.stderr or created.stdout or "docker create failed").strip()
+            raise IsolationUnavailable(runtime, detail)
+        container_id = created.stdout.strip()
+        started = self._docker(["start", sandbox_id], timeout=60)
+        if started.returncode != 0:
+            self._docker(["rm", "-f", sandbox_id], timeout=30)
+            detail = (started.stderr or started.stdout or "docker start failed").strip()
+            raise IsolationUnavailable(runtime, detail)
+        return SandboxHandle(

+            container_id=container_id,
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            runtime=runtime,
+            retain=retain,
+            status="running",

+        return self._load(sandbox_id, tenant_id=tenant_id)
+
+    def stop(self, sandbox_id: str, *, tenant_id: str) -> SandboxHandle:
+        self._load(sandbox_id, tenant_id=tenant_id)
+        stopped = self._docker(["stop", "-t", "1", sandbox_id], timeout=30)
+        if stopped.returncode != 0:
+            detail = (stopped.stderr or stopped.stdout or "docker stop failed").strip()
+            raise IsolationUnavailable(sandbox_id, detail)
+        return self._load(sandbox_id, tenant_id=tenant_id)
+
+    def destroy(self, sandbox_id: str, *, tenant_id: str) -> None:
+        self._load(sandbox_id, tenant_id=tenant_id)
+        removed = self._docker(["rm", "-f", sandbox_id], timeout=30)
+        if removed.returncode != 0:
+            detail = (removed.stderr or removed.stdout or "docker rm failed").strip()
+            raise IsolationUnavailable(sandbox_id, detail)
+
+    def health(self, sandbox_id: str, *, tenant_id: str) -> dict[str, str]:
+        handle = self._load(sandbox_id, tenant_id=tenant_id)
+        return {
+            "status": handle.status,
+            "sandbox_id": handle.sandbox_id,
+            "runtime": handle.runtime,
+            "workspace_id": handle.workspace_id,
+        }
+
+    def exec(
+        self,
+        sandbox_id: str,
+        *,
+        tenant_id: str,
+        argv: list[str],
+        timeout: float = 60,
+    ) -> ExecResult:
+        handle = self._load(sandbox_id, tenant_id=tenant_id)
+        if handle.status != "running":
+            raise IsolationUnavailable(sandbox_id, "sandbox is not running")
+        if not argv or not all(isinstance(part, str) and part for part in argv):
+            raise IsolationUnavailable(sandbox_id, "tool argv is empty")
+        proc = self._docker(["exec", sandbox_id, *argv], timeout=timeout)
+        return ExecResult(
+            sandbox_id=handle.sandbox_id,
+            container_id=handle.container_id,
+            exit_code=proc.returncode,
+            stdout=proc.stdout,
+            stderr=proc.stderr,
+        )
+
+    def _select_runtime(self) -> str:
+        names = set(self.installed_runtimes())
+        if self._runtime:
+            if self._runtime not in names:
+                raise IsolationUnavailable(
+                    self._runtime,
+                    f"runtime class {self._runtime} is not installed",
+                )
+            return self._runtime
+        if "runsc" in names:
+            return "runsc"
+        if "runc" in names:
+            return "runc"
+        raise IsolationUnavailable("runtime", "no isolation runtime is installed")
+
+    def _load(self, sandbox_id: str, *, tenant_id: str) -> SandboxHandle:
+        _require_id(tenant_id)
+        if not _SANDBOX.fullmatch(sandbox_id):
+            raise AttachRefused(sandbox_id, tenant_id)
+        proc = self._docker(["inspect", sandbox_id], timeout=30)
+        if proc.returncode != 0 or not proc.stdout.strip():
+            raise AttachRefused(sandbox_id, tenant_id)
+        try:
+            info = json.loads(proc.stdout)[0]
+        except (json.JSONDecodeError, IndexError, TypeError) as exc:
+            raise AttachRefused(sandbox_id, tenant_id) from exc
+        labels = (info.get("Config") or {}).get("Labels") or {}
+        owner = str(labels.get("rxy.tenant", ""))
+        if owner != tenant_id:
+            raise AttachRefused(sandbox_id, tenant_id)
+        state = str((info.get("State") or {}).get("Status", ""))
+        if state == "running":
+            status = "running"
+        elif state in {"exited", "dead", "created"}:
+            status = "stopped"
+        else:
+            status = "stopped"
+        runtime = str((info.get("HostConfig") or {}).get("Runtime") or "")
+        return SandboxHandle(
+            sandbox_id=str(labels.get("rxy.sandbox") or sandbox_id),
+            container_id=str(info.get("Id") or ""),
+            tenant_id=owner,
+            workspace_id=str(labels.get("rxy.workspace") or ""),
+            runtime=runtime,
+            retain=str(labels.get("rxy.retain") or "") == "true",
+            status=status,
+        )
+
+    def _docker(self, args: list[str], *, timeout: float) -> subprocess.CompletedProcess[str]:
+        try:
+            return subprocess.run(
+                [self._docker_bin, *args],
+                capture_output=True,
+                text=True,
+                timeout=timeout,
+                check=False,
+            )
+        except subprocess.TimeoutExpired as exc:
+            raise IsolationUnavailable(args[0], "docker command timed out") from exc
+        except OSError as exc:
+            raise IsolationUnavailable("docker", str(exc)) from exc
+
+
+def _require_id(value: str) -> None:
+    if not _ID.fullmatch(value):
+        raise IsolationUnavailable(value, "tenant or workspace id is refused")

+"""Replaceable cloud sandbox types.
+
+The provider life cycle is create, attach, stop, destroy, and health. It is
+not a Run. A missing isolation runtime fails closed.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from typing import Protocol
+
+
+class AttachRefused(PermissionError):
+    """Another tenant, or an unknown sandbox, cannot attach."""
+
+    def __init__(self, sandbox_id: str, tenant_id: str) -> None:
+        self.sandbox_id = sandbox_id
+        self.tenant_id = tenant_id
+        super().__init__(f"attach refused: {tenant_id} cannot attach {sandbox_id}")
+
+
+class IsolationUnavailable(RuntimeError):
+    """The named target has no isolation capability."""
+
+    def __init__(self, target: str, detail: str) -> None:
+        self.target = target
+        self.detail = detail
+        super().__init__(f"target unavailable: {target}: {detail}")
+
+
+@dataclass(frozen=True)
+class SandboxHandle:
+    sandbox_id: str
+    container_id: str
+    tenant_id: str
+    workspace_id: str
+    runtime: str
+    retain: bool
+    status: str
+
+
+@dataclass(frozen=True)
+class ExecResult:
+    sandbox_id: str
+    container_id: str
+    exit_code: int
+    stdout: str
+    stderr: str
+
+
+class SandboxProvider(Protocol):
+    """One isolation backend. A VM provider can replace the container adapter."""
+
+    def create(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        retain: bool = True,
+    ) -> SandboxHandle:
+        """Start an isolated sandbox. This does not open a Run."""
+
+    def attach(self, sandbox_id: str, *, tenant_id: str) -> SandboxHandle:
+        """Return the sandbox only when the caller owns it."""
+
+    def stop(self, sandbox_id: str, *, tenant_id: str) -> SandboxHandle:
+        """Stop the sandbox and keep its record."""
+
+    def destroy(self, sandbox_id: str, *, tenant_id: str) -> None:
+        """Remove the sandbox."""
+
+    def health(self, sandbox_id: str, *, tenant_id: str) -> dict[str, str]:
+        """Report status for the owning tenant."""
+
+    def exec(
+        self,
+        sandbox_id: str,
+        *,
+        tenant_id: str,
+        argv: list[str],
+        timeout: float = 60,
+    ) -> ExecResult:
+        """Run argv inside the sandbox. Never on the server host."""

+"""Tenant to workspace map and the sandboxd tool endpoint.
+
+A Run id only records billing. The sandbox stays up when retain is set, and
+close destroys it otherwise. An unavailable runtime blocks the tool.
+"""
+
+from __future__ import annotations
+
+import time
+import uuid
+
+from rxycode_runner.sandbox.provider import (
+    AttachRefused,
+    ExecResult,
+    IsolationUnavailable,
+    SandboxHandle,
+    SandboxProvider,
+)
+
+
+class ToolBlocked(RuntimeError):
+    """The tool did not start. The target is unavailable."""
+
+    def __init__(self, target: str, detail: str) -> None:
+        self.target = target
+        self.detail = detail
+        super().__init__(f"tool blocked: target unavailable: {target}: {detail}")
+
+
+class SandboxControl:
+    """Maps each tenant to its workspaces and talks to one SandboxProvider."""
+
+    def __init__(self, provider: SandboxProvider) -> None:
+        self._provider = provider
+        self._workspaces: dict[str, dict[str, str]] = {}
+        self._runs: dict[str, dict[str, object]] = {}
+        self._started: dict[str, float] = {}
+        self._ledger: list[dict[str, object]] = []
+
+    def derive_target(self, tenant_id: str, workspace_id: str) -> str:
+        return f"cloud:{tenant_id}:{workspace_id}"
+
+    def workspaces(self, tenant_id: str) -> dict[str, str]:
+        return dict(self._workspaces.get(tenant_id, {}))
+
+    def ledger(self) -> list[dict[str, object]]:
+        return [dict(row) for row in self._ledger]
+
+    def create_sandbox(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        retain: bool = True,
+    ) -> SandboxHandle:
+        target = self.derive_target(tenant_id, workspace_id)
+        try:
+            handle = self._provider.create(
+                tenant_id=tenant_id,
+                workspace_id=workspace_id,
+                retain=retain,
+            )
+        except IsolationUnavailable as exc:
+            self._record(
+                "blocked",
+                sandbox_id="",
+                tenant_id=tenant_id,
+                workspace_id=workspace_id,
+                target=target,
+                runtime_seconds=0.0,
+            )
+            raise ToolBlocked(target, str(exc)) from exc
+        self._workspaces.setdefault(tenant_id, {})[handle.sandbox_id] = workspace_id
+        self._started[handle.sandbox_id] = time.monotonic()
+        self._record(
+            "start",
+            sandbox_id=handle.sandbox_id,
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            target=target,
+            runtime_seconds=0.0,
+        )
+        return handle
+
+    def sandboxd_exec(
+        self,
+        *,
+        tenant_id: str,
+        argv: list[str],
+        sandbox_id: str | None = None,
+        workspace_id: str | None = None,
+        timeout: float = 60,
+    ) -> ExecResult:
+        if sandbox_id is None:
+            if not workspace_id:
+                raise ToolBlocked("cloud:missing", "workspace is required")
+            created = self.create_sandbox(
+                tenant_id=tenant_id,
+                workspace_id=workspace_id,
+                retain=True,
+            )
+            sandbox_id = created.sandbox_id
+            workspace_id = created.workspace_id
+        try:
+            return self._provider.exec(
+                sandbox_id,
+                tenant_id=tenant_id,
+                argv=argv,
+                timeout=timeout,
+            )
+        except IsolationUnavailable as exc:
+            workspace = workspace_id or self._workspace_of(tenant_id, sandbox_id)
+            target = self.derive_target(tenant_id, workspace or "unknown")
+            self._record(
+                "blocked",
+                sandbox_id=sandbox_id,
+                tenant_id=tenant_id,
+                workspace_id=workspace or "",
+                target=target,
+                runtime_seconds=0.0,
+            )
+            raise ToolBlocked(target, str(exc)) from exc
+
+    def open_run(self, *, tenant_id: str, sandbox_id: str) -> str:
+        handle = self._provider.attach(sandbox_id, tenant_id=tenant_id)
+        run_id = "run-" + uuid.uuid4().hex[:12]
+        self._runs[run_id] = {
+            "tenant_id": tenant_id,
+            "sandbox_id": sandbox_id,
+            "workspace_id": handle.workspace_id,
+            "retain": handle.retain,
+        }
+        return run_id
+
+    def close_run(self, *, tenant_id: str, run_id: str) -> None:
+        row = self._runs.get(run_id)
+        if row is None or row["tenant_id"] != tenant_id:
+            raise AttachRefused(run_id, tenant_id)
+        sandbox_id = str(row["sandbox_id"])
+        workspace_id = str(row["workspace_id"])
+        self._record(
+            "bill",
+            sandbox_id=sandbox_id,
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            target=self.derive_target(tenant_id, workspace_id),
+            runtime_seconds=self._elapsed(sandbox_id),
+        )
+        del self._runs[run_id]
+        if row["retain"] is not True:
+            self.destroy(tenant_id=tenant_id, sandbox_id=sandbox_id)
+
+    def stop(self, *, tenant_id: str, sandbox_id: str) -> None:
+        self._provider.stop(sandbox_id, tenant_id=tenant_id)
+
+    def destroy(self, *, tenant_id: str, sandbox_id: str) -> None:
+        handle = self._provider.attach(sandbox_id, tenant_id=tenant_id)
+        self._provider.destroy(sandbox_id, tenant_id=tenant_id)
+        self._workspaces.get(tenant_id, {}).pop(sandbox_id, None)
+        self._record(
+            "destroy",
+            sandbox_id=sandbox_id,
+            tenant_id=tenant_id,
+            workspace_id=handle.workspace_id,
+            target=self.derive_target(tenant_id, handle.workspace_id),
+            runtime_seconds=self._elapsed(sandbox_id),
+        )
+
+    def health(self, *, tenant_id: str, sandbox_id: str) -> dict[str, str]:
+        return self._provider.health(sandbox_id, tenant_id=tenant_id)
+
+    def _workspace_of(self, tenant_id: str, sandbox_id: str) -> str:
+        return self._workspaces.get(tenant_id, {}).get(sandbox_id, "")
+
+    def _elapsed(self, sandbox_id: str) -> float:
+        started = self._started.get(sandbox_id)
+        if started is None:
+            return 0.0
+        return max(0.0, time.monotonic() - started)
+
+    def _record(
+        self,
+        kind: str,
+        *,
+        sandbox_id: str,
+        tenant_id: str,
+        workspace_id: str,
+        target: str,
+        runtime_seconds: float,
+    ) -> None:
+        self._ledger.append(
+            {
+                "kind": kind,
+                "sandbox_id": sandbox_id,
+                "tenant_id": tenant_id,
+                "workspace_id": workspace_id,
+                "target": target,
+                "runtime_seconds": runtime_seconds,
+            }
+        )
