+export type RunEvent = {
+  seq: number
+  type: string
+  payload: Record<string, unknown>
+}
+
+export type ToolCard = {
+  callId: string
+  name: string
+  status: "failed" | "completed"
+}
+
+export type ApprovalCard = {
+  id: string
+  status: string
+}
+
+export type Projection = {
+  order: number[]
+  answer: string
+  intermediate: string[]
+  tools: ToolCard[]
+  approvals: ApprovalCard[]
+}
+
+function textOf(payload: Record<string, unknown>): string {
+  return String(payload.text ?? "")
+}
+
+export function reduceEvents(events: RunEvent[]): Projection {
+  const seen = new Set<number>()
+  const pending = new Map<number, RunEvent>()
+  let next = 1
+  const projection: Projection = {
+    order: [],
+    answer: "",
+    intermediate: [],
+    tools: [],
+    approvals: [],
+  }
+
+  const apply = (event: RunEvent) => {
+    projection.order.push(event.seq)
+    const payload = event.payload ?? {}
+    if (event.type === "assistant.final") {
+      projection.answer = textOf(payload)
+      return
+    }
+    if (event.type === "assistant.delta" && payload.intermediate === true) {
+      projection.intermediate.push(textOf(payload))
+      return
+    }
+    if (event.type === "tool.failed" || event.type === "tool.completed") {
+      projection.tools.push({
+        callId: String(payload.call_id ?? ""),
+        name: String(payload.tool ?? ""),
+        status: event.type === "tool.failed" ? "failed" : "completed",
+      })
+      return
+    }
+    if (event.type === "approval.requested") {
+      projection.approvals.push({
+        id: String(payload.approval_id ?? ""),
+        status: "pending",
+      })
+      return
+    }
+    if (event.type === "approval.resolved") {
+      const id = String(payload.approval_id ?? "")
+      const status = String(payload.status ?? "")
+      const found = projection.approvals.find((item) => item.id === id)
+      if (found) found.status = status
+      else projection.approvals.push({ id, status })
+    }
+  }
+
+  const flush = () => {
+    while (pending.has(next)) {
+      const event = pending.get(next)
+      pending.delete(next)
+      if (event) apply(event)
+      next += 1
+    }
+  }
+
+  for (const event of events) {
+    if (!Number.isInteger(event.seq) || event.seq < 1 || seen.has(event.seq)) continue
+    seen.add(event.seq)
+    if (event.seq > next) {
+      pending.set(event.seq, event)
+      continue
+    }
+    apply(event)
+    next += 1
+    flush()
+  }
+  return projection
+}

+import { useEffect, useState } from "react"
+
+import { reduceEvents, type Projection, type RunEvent } from "./eventReducer"
+
+export type LiveRun = {
+  projection: Projection
+  cursorExpired: boolean
+  snapshotSeq: number | null
+  connected: boolean
+}
+
+function parseFrames(text: string): RunEvent[] {
+  const events: RunEvent[] = []
+  for (const block of text.split("\n\n")) {
+    const dataLine = block.split("\n").find((line) => line.startsWith("data:"))
+    if (!dataLine) continue
+    try {
+      const body = JSON.parse(dataLine.slice(5).trim()) as RunEvent & { code?: string }
+      if (body.code || typeof body.seq !== "number") continue
+      events.push({ seq: body.seq, type: body.type, payload: body.payload ?? {} })
+    } catch {
+      continue
+    }
+  }
+  return events
+}
+
+export function useRunEvents(runId: string): LiveRun {
+  const [events, setEvents] = useState<RunEvent[]>([])
+  const [cursorExpired, setCursorExpired] = useState(false)
+  const [snapshotSeq, setSnapshotSeq] = useState<number | null>(null)
+  const [connected, setConnected] = useState(false)
+
+  useEffect(() => {
+    let source: EventSource | null = null
+    let stopped = false
+    const load = async () => {
+      const path = `/api/v1/runs/${encodeURIComponent(runId)}/events?after_seq=0&follow=0`
+      const response = await fetch(path, {
+        credentials: "same-origin",
+        headers: { accept: "text/event-stream" },
+      })
+      if (stopped) return
+      if (response.status === 409) {
+        const body = (await response.json()) as { code?: string; snapshot_seq?: number }
+        if (body.code === "CURSOR_EXPIRED") {
+          setCursorExpired(true)
+          setSnapshotSeq(typeof body.snapshot_seq === "number" ? body.snapshot_seq : null)
+          setEvents([])
+        }
+        return
+      }
+      if (!response.ok) return
+      const snapshot = parseFrames(await response.text())
+      if (stopped) return
+      setEvents(snapshot)
+      setConnected(true)
+      source = new EventSource(`/api/v1/runs/${encodeURIComponent(runId)}/events?follow=1`)
+      source.onmessage = (message) => {
+        try {
+          const body = JSON.parse(message.data) as RunEvent
+          if (typeof body.seq !== "number") return
+          setEvents((current) => [...current, { ...body, payload: body.payload ?? {} }])
+        } catch {
+          return
+        }
+      }
+      source.addEventListener("error", (event) => {
+        const data = (event as MessageEvent).data
+        if (typeof data !== "string") {
+          setConnected(false)
+          return
+        }
+        try {
+          const body = JSON.parse(data) as { code?: string; snapshot_seq?: number }
+          if (body.code !== "CURSOR_EXPIRED") return
+          setCursorExpired(true)
+          setSnapshotSeq(typeof body.snapshot_seq === "number" ? body.snapshot_seq : null)
+          setEvents([])
+          source?.close()
+        } catch {
+          setConnected(false)
+        }
+      })
+    }
+    void load()
+    return () => {
+      stopped = true
+      source?.close()
+    }
+  }, [runId])
+
+  return {
+    projection: reduceEvents(events),
+    cursorExpired,
+    snapshotSeq,
+    connected,
+  }
+}

+    proc = subprocess.run([_bun(), str(driver)], capture_output=True, text=True, check=False)
+    assert proc.returncode == 0, proc.stderr or proc.stdout
