+"""Checkout and cleanup inside a sandbox. The API host does not run git."""
+
+from __future__ import annotations
+
+import base64
+import subprocess
+import tempfile
+import uuid
+from pathlib import Path

+from rxycode_runner.sandbox.gvisor import docker_executable
+from rxycode_runner.sandbox.provider import SandboxHandle

+_TREE = (
+    "import base64, os, sys\n"
+    "root = '/tmp/ws'\n"
+    "chunks = []\n"
+    "for dirpath, dirs, files in os.walk(root):\n"
+    "    dirs[:] = [name for name in sorted(dirs) if name != '.git']\n"
+    "    for name in sorted(files):\n"
+    "        path = os.path.join(dirpath, name)\n"
+    "        rel = os.path.relpath(path, root).replace('\\\\', '/')\n"
+    "        data = open(path, 'rb').read()\n"
+    "        chunks.append(rel.encode() + b'\\0' + data + b'\\0')\n"
+    "sys.stdout.write(base64.b64encode(b''.join(chunks)).decode())\n"
+)

+class CheckoutFailed(RuntimeError):
+    """The sandbox git checkout did not land on the requested revision."""
+
+
+def prepare_checkout(*, bundle: bytes, revision: str) -> str:
+    """Clone and detach HEAD in a network-none git sandbox. Returns its name."""
+    docker = docker_executable()
+    git_name = "rxy-git-" + uuid.uuid4().hex[:12]
+    bundle_path = _write_bundle(bundle)
+    try:
+        created = _run(
+            docker,
+            [
+                "create",
+                "--name",
+                git_name,
+                "--entrypoint",
+                "sleep",
+                "--user",
+                "65534:65534",
+                "--network",
+                "none",
+                "--cap-drop",
+                "ALL",
+                "--security-opt",
+                "no-new-privileges",
+                "--workdir",
+                "/tmp",
+                "alpine/git",
+                "infinity",
+            ],
+        )
+        if created.returncode != 0:
+            raise CheckoutFailed((created.stderr or created.stdout or "git sandbox create failed").strip())
+        if _run(docker, ["start", git_name]).returncode != 0:
+            raise CheckoutFailed("git sandbox failed to start")
+        if _run(docker, ["cp", bundle_path, f"{git_name}:/tmp/sample.bundle"]).returncode != 0:
+            raise CheckoutFailed("bundle copy failed")
+        cloned = _exec(docker, git_name, ["git", "clone", "--quiet", "/tmp/sample.bundle", "/tmp/ws"])
+        if cloned.returncode != 0:
+            raise CheckoutFailed((cloned.stderr or "git clone failed").strip())
+        checked = _exec(
+            docker,
+            git_name,
+            ["git", "-C", "/tmp/ws", "checkout", "--detach", revision],
+        )
+        if checked.returncode != 0:
+            raise CheckoutFailed((checked.stderr or "git checkout failed").strip())
+        head = _exec(docker, git_name, ["git", "-C", "/tmp/ws", "rev-parse", "HEAD"])
+        if head.returncode != 0 or head.stdout.strip() != revision:
+            raise CheckoutFailed("sandbox revision does not match")
+        marked = _exec(
+            docker,
+            git_name,
+            [
+                "sh",
+                "-c",
+                "git -C /tmp/ws status --porcelain > /tmp/clean-status && "
+                "git -C /tmp/ws rev-parse HEAD > /tmp/ws/.revision && "
+                "cp /tmp/clean-status /tmp/ws/.porcelain && "
+                "git -C /tmp/ws rev-parse --abbrev-ref HEAD > /tmp/ws/.branch",
+            ],
+        )
+        if marked.returncode != 0:
+            raise CheckoutFailed((marked.stderr or "git status failed").strip())
+        return git_name
+    except Exception:
+        _run(docker, ["rm", "-f", git_name])
+        raise
+    finally:
+        Path(bundle_path).unlink(missing_ok=True)
+
+
+def adopt_checkout(handle: SandboxHandle, git_name: str) -> None:
+    """Copy the checked-out tree into the tenant sandbox. Does not run host git."""
+    docker = docker_executable()
+    made = _exec(
+        docker,
+        handle.sandbox_id,
+        ["python", "-c", "import os; os.makedirs('/tmp/ws', exist_ok=True)"],
+    )
+    if made.returncode != 0:
+        raise CheckoutFailed((made.stderr or "workspace dir failed").strip())
+    source = subprocess.Popen(
+        [docker, "exec", git_name, "tar", "-C", "/tmp/ws", "-cf", "-", "."],
+        stdout=subprocess.PIPE,
+        stderr=subprocess.PIPE,
+    )
+    assert source.stdout is not None
+    dest = subprocess.run(
+        [docker, "exec", "-i", handle.sandbox_id, "tar", "-C", "/tmp/ws", "-xf", "-"],
+        stdin=source.stdout,
+        capture_output=True,
+        timeout=60,
+        check=False,
+    )
+    source.stdout.close()
+    err = source.stderr.read() if source.stderr is not None else b""
+    code = source.wait(timeout=60)
+    if code != 0 or dest.returncode != 0:
+        detail = (dest.stderr or err or b"workspace copy failed").decode("utf-8", "replace")
+        raise CheckoutFailed(detail.strip())
+
+
+def discard_checkout(git_name: str) -> None:
+    _run(docker_executable(), ["rm", "-f", git_name])
+
+
+def run_in_workspace(handle: SandboxHandle, argv: list[str]) -> tuple[int, str, str]:
+    done = _exec(docker_executable(), handle.sandbox_id, argv)
+    return done.returncode, done.stdout, done.stderr
+
+
+def tree_bytes(handle: SandboxHandle) -> bytes:
+    code, out, err = run_in_workspace(handle, ["python", "-c", _TREE])
+    if code != 0:
+        raise CheckoutFailed((err or "tree archive failed").strip())
+    return base64.b64decode(out.strip())
+
+
+def _write_bundle(bundle: bytes) -> str:
+    handle = tempfile.NamedTemporaryFile(prefix="rxy-bundle-", suffix=".bundle", delete=False)
+    try:
+        handle.write(bundle)
+    finally:
+        handle.close()
+    return handle.name
+
+
+def _exec(docker: str, name: str, argv: list[str]) -> subprocess.CompletedProcess[str]:
+    return _run(docker, ["exec", name, *argv])
+
+
+def _run(docker: str, args: list[str]) -> subprocess.CompletedProcess[str]:
+    return subprocess.run(
+        [docker, *args],
+        capture_output=True,
+        text=True,
+        timeout=60,
+        check=False,
+    )

+_REPO = re.compile(r"^https://github\.com/[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+(?:\.git)?$")
+_REV = re.compile(r"^[0-9a-f]{40}$")

+
+class RepoRefused(PermissionError):
+    """The repo URL or revision is outside the allowlist."""
+
+    def __init__(self, detail: str = "repo url refused") -> None:
+        super().__init__(detail)
+
+
+@dataclass(frozen=True)
+class ImportedWorkspace:
+    workspace_id: str
+    sandbox_id: str
+    container_id: str
+    revision: str
+    test_stdout: str
+    log: str
+    snapshot_id: str
+    snapshot_sha256: str
+
+
+class WorkspaceProvision:
+    def __init__(self) -> None:
+        self._store = SnapshotStore()
+        self._rows: dict[str, dict[str, str]] = {}
+
+    def import_repo(
+        self,
+        *,
+        tenant_id: str,
+        project_id: str,
+        url: str,
+        revision: str,
+        bundle: bytes,
+        token: str | None = None,
+        authorize_snapshot: bool = False,
+        ttl_seconds: int | None = None,
+    ) -> ImportedWorkspace:
+        redacted = _redact(url, token)
+        if not _REPO.fullmatch(redacted) or ".." in redacted:
+            raise RepoRefused("repo url refused")
+        if not _REV.fullmatch(revision):
+            raise RepoRefused("revision refused")
+        if not bundle:
+            raise RepoRefused("bundle refused")
+        policy = load_policy()
+        if ttl_seconds is not None:
+            policy = replace(policy, ttl_seconds=ttl_seconds)
+        git_name = prepare_checkout(bundle=bundle, revision=revision)
+        handle = None
+        try:
+            handle = launch(
+                tenant_id=tenant_id,
+                workspace_id=project_id,
+                policy=policy,
+            )
+            adopt_checkout(handle, git_name)
+            code, stdout, stderr = run_in_workspace(
+                handle,
+                ["python", "/tmp/ws/test_sample.py"],
+            )
+            if code != 0:
+                raise CheckoutFailed(_redact(stderr or "sandbox test failed", token))
+            blob = tree_bytes(handle)
+            blob = _redact_bytes(blob, token)
+            snapshot_id, digest = self._store.put(
+                tenant_id=tenant_id,
+                blob=blob,
+                authorized=authorize_snapshot,
+            )
+            workspace_id = "ws-" + uuid.uuid4().hex[:12]
+            self._rows[workspace_id] = {
+                "tenant_id": tenant_id,
+                "sandbox_id": handle.sandbox_id,
+                "container_id": handle.container_id,
+            }
+            log = _redact(f"imported {redacted} revision {revision}", token)
+            return ImportedWorkspace(
+                workspace_id=workspace_id,
+                sandbox_id=handle.sandbox_id,
+                container_id=handle.container_id,
+                revision=revision,
+                test_stdout=_redact(stdout, token),
+                log=log,
+                snapshot_id=snapshot_id,
+                snapshot_sha256=digest,
+            )
+        except Exception:
+            if handle is not None:
+                GVisorSandbox().destroy(handle.sandbox_id, tenant_id=tenant_id)
+            raise
+        finally:
+            discard_checkout(git_name)
+
+    def read_snapshot(self, *, tenant_id: str, snapshot_id: str) -> bytes:
+        return self._store.read(tenant_id=tenant_id, snapshot_id=snapshot_id)
+
+    def cleanup(self, *, tenant_id: str, workspace_id: str) -> None:
+        row = self._rows.get(workspace_id)
+        if row is None or row["tenant_id"] != tenant_id:
+            raise SnapshotRefused(workspace_id, tenant_id)
+        GVisorSandbox().destroy(row["sandbox_id"], tenant_id=tenant_id)
+        del self._rows[workspace_id]
+
+
+def _redact(text: str, token: str | None = None) -> str:
+    cleaned = re.sub(r"(https?://)[^/@\s]+@", r"\1", text)
+    if token:
+        cleaned = cleaned.replace(token, "[redacted]")
+    return cleaned
+
+
+def _redact_bytes(blob: bytes, token: str | None) -> bytes:
+    if not token:
+        return blob
+    return blob.replace(token.encode(), b"[redacted]")

+"""Snapshot bytes stay with the owning tenant. Unauthorized snapshots are not kept."""
+
+from __future__ import annotations
+
+import hashlib
+import uuid
+
+
+class SnapshotRefused(PermissionError):
+    def __init__(self, snapshot_id: str, tenant_id: str) -> None:
+        self.snapshot_id = snapshot_id
+        self.tenant_id = tenant_id
+        super().__init__(f"snapshot refused: {tenant_id} cannot read {snapshot_id}")
+
+
+class SnapshotStore:
+    def __init__(self) -> None:
+        self._rows: dict[str, dict[str, object]] = {}
+
+    def put(self, *, tenant_id: str, blob: bytes, authorized: bool) -> tuple[str, str]:
+        digest = hashlib.sha256(blob).hexdigest()
+        if not authorized:
+            return "", digest
+        snapshot_id = "snap-" + uuid.uuid4().hex[:12]
+        self._rows[snapshot_id] = {"tenant_id": tenant_id, "blob": blob}
+        return snapshot_id, digest
+
+    def read(self, *, tenant_id: str, snapshot_id: str) -> bytes:
+        row = self._rows.get(snapshot_id)
+        if row is None or row["tenant_id"] != tenant_id:
+            raise SnapshotRefused(snapshot_id, tenant_id)
+        blob = row["blob"]
+        assert isinstance(blob, bytes)
+        return blob

+                timeout=30,
+                check=False,
+            )
+            if inspected.returncode != 0:
+                status = "gone"
+                break
+            info = json.loads(inspected.stdout)[0]
+            state = info["State"]["Status"]
+            if state in {"exited", "dead"}:
+                status = "stopped"
+                break
+            time.sleep(0.3)
+        assert status in {"stopped", "gone"}
+        provision.cleanup(tenant_id="tenant-a", workspace_id=short.workspace_id)
+        short = None
+    finally:
+        if imported is not None:
+            _remove(docker, imported.container_id)
+        if short is not None:
+            _remove(docker, short.container_id)

+def test_w5_c03_negative(tmp_path: Path) -> None:
+    """A private-repo token stays out of the log and the snapshot. Other tenants cannot read it."""
+    from rxycode_runner.sandbox.gvisor import docker_executable
+    from rxycode_server.workspaces.provision import RepoRefused, SnapshotRefused, WorkspaceProvision
+
+    docker = docker_executable()
+    sha, bundle = _sample_bundle(tmp_path)
+    secret = "w5c03-token-" + uuid.uuid4().hex
+    poisoned = f"https://x-access-token:{secret}@github.com/rxycode/private.git"
+    provision = WorkspaceProvision()
+    before = subprocess.run(
+        [docker, "ps", "-aq", "--filter", "label=rxy.card=w5c01"],
+        capture_output=True,
+        text=True,
+        timeout=30,
+        check=False,
+    )
+    before_ids = {line.strip() for line in before.stdout.splitlines() if line.strip()}
+    for url in (
+        "file:///etc/passwd",
+        "https://169.254.169.254/latest/meta-data",
+        "https://github.com/../../etc",
+        "http://evil.example/repo.git",
+        f"https://x-access-token:{secret}@evil.example/repo.git",
+    ):
+        with pytest.raises(RepoRefused) as refused:
+            provision.import_repo(
+                tenant_id="tenant-a",
+                project_id="proj-bad",
+                url=url,
+                revision=sha,
+                token=secret,
+                bundle=bundle,
+            )
+        assert secret not in str(refused.value)
+    after = subprocess.run(
+        [docker, "ps", "-aq", "--filter", "label=rxy.card=w5c01"],
+        capture_output=True,
+        text=True,
+        timeout=30,
+        check=False,
+    )
+    after_ids = {line.strip() for line in after.stdout.splitlines() if line.strip()}
+    assert after_ids == before_ids
+    imported = None
+    try:
+        imported = provision.import_repo(
+            tenant_id="tenant-a",
+            project_id="proj-private",
+            url=poisoned,
+            revision=sha,
+            token=secret,
+            bundle=bundle,
+            authorize_snapshot=True,
+        )
+        assert secret not in imported.log
+        assert secret not in imported.revision
+        blob = provision.read_snapshot(tenant_id="tenant-a", snapshot_id=imported.snapshot_id)
+        assert secret.encode() not in blob
+        inspected = subprocess.run(
+            [docker, "inspect", imported.container_id],
+            capture_output=True,
+            text=True,
+            timeout=30,
+            check=False,
+        )
+        assert inspected.returncode == 0, inspected.stderr
+        assert secret not in inspected.stdout
+        with pytest.raises(SnapshotRefused):
+            provision.read_snapshot(tenant_id="tenant-b", snapshot_id=imported.snapshot_id)
+        with pytest.raises(SnapshotRefused):
+            provision.cleanup(tenant_id="tenant-b", workspace_id=imported.workspace_id)
+        assert _read(docker, imported.container_id, "/tmp/ws/base.txt") == "fixed-base\n"
+        provision.cleanup(tenant_id="tenant-a", workspace_id=imported.workspace_id)
+        imported = None
+    finally:
+        if imported is not None:
+            _remove(docker, imported.container_id)
