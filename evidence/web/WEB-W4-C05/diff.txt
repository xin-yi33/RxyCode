+"""Local Remote Full grants. The cloud permission string is not authority."""
+
+from __future__ import annotations
+
+import json
+import time
+from collections.abc import Callable
+from pathlib import Path
+from typing import Any
+from uuid import uuid4
+
+_STORES: dict[str, "GrantStore"] = {}
+_SENDERS: dict[str, Callable[[tuple[int, int]], None]] = {}
+
+
+class GrantStore:
+    """Grants this machine confirmed. Expiry and revoke are checked on each call."""
+
+    def __init__(self, directory: Path, clock: Callable[[], float] | None = None) -> None:
+        self.directory = Path(directory)
+        self.directory.mkdir(parents=True, exist_ok=True)
+        self._clock = clock or time.time
+
+    def clock(self) -> float:
+        return float(self._clock())
+
+    def confirmed_count(self) -> int:
+        return sum(1 for item in self._read_grants() if item.get("confirmed") is True)
+
+    def latest_grant_id(self) -> str:
+        confirmed = [item for item in self._read_grants() if item.get("confirmed") is True]
+        if not confirmed:
+            raise LookupError("no confirmed grant")
+        return str(confirmed[-1]["grant_id"])
+
+    def confirm(self, request: dict[str, Any]) -> dict[str, Any]:
+        expires_at = float(request["expires_at"])
+        if self.clock() >= expires_at:
+            raise PermissionError("GRANT_EXPIRED")
+        record = {
+            "grant_id": "gr_" + uuid4().hex,
+            "device_id": str(request.get("device_id") or ""),
+            "session_id": str(request.get("session_id") or ""),
+            "run_id": str(request.get("run_id") or ""),
+            "workspace": str(request.get("workspace") or ""),
+            "capabilities": [str(item) for item in request.get("capabilities") or []],
+            "expires_at": expires_at,
+            "revoke_epoch": self._epoch(),
+            "confirmed": True,
+        }
+        grants = self._read_grants()
+        grants.append(record)
+        self._write_grants(grants)
+        return record
+
+    def revoke(self, grant_id: str) -> int:
+        epoch = self._epoch() + 1
+        (self.directory / "epoch.txt").write_text(str(epoch), encoding="utf-8")
+        grants = self._read_grants()
+        for item in grants:
+            if item.get("grant_id") == grant_id:
+                item["confirmed"] = False
+        self._write_grants(grants)
+        return epoch
+
+    def evaluate(self, call: dict[str, Any]) -> tuple[bool, str]:
+        reason = self._reason(call)
+        self._audit(
+            {
+                "permission": str(call.get("permission") or ""),
+                "allowed": reason == "",
+                "reason": reason or "granted",
+                "session_id": str(call.get("session_id") or ""),
+                "run_id": str(call.get("run_id") or ""),
+                "capability": str(call.get("capability") or ""),
+                "workspace": str(call.get("workspace") or ""),
+            }
+        )
+        return reason == "", reason
+
+    def audit_entries(self) -> list[dict[str, Any]]:
+        path = self.directory / "audit.jsonl"
+        if not path.exists():
+            return []
+        rows: list[dict[str, Any]] = []
+        for line in path.read_text(encoding="utf-8").splitlines():
+            if line.strip():
+                rows.append(json.loads(line))
+        return rows
+
+    def _reason(self, call: dict[str, Any]) -> str:
+        grants = self._read_grants()
+        if not grants:
+            return "GRANT_REQUIRED"
+        session_id = str(call.get("session_id") or "")
+        matched = [item for item in grants if item.get("session_id") == session_id]
+        if not matched:
+            return "SESSION_MISMATCH"
+        grant = matched[-1]
+        stored_epoch = grant.get("revoke_epoch")
+        if stored_epoch is None or int(stored_epoch) != self._epoch() or grant.get("confirmed") is not True:
+            return "GRANT_REVOKED"
+        if self.clock() >= float(grant.get("expires_at") or 0):
+            return "GRANT_EXPIRED"
+        if str(call.get("run_id") or "") != str(grant.get("run_id") or ""):
+            return "RUN_MISMATCH"
+        capability = str(call.get("capability") or "")
+        if capability not in {str(item) for item in grant.get("capabilities") or []}:
+            return "CAPABILITY_MISMATCH"
+        root = str(grant.get("workspace") or "")
+        workspace = str(call.get("workspace") or "")
+        workdir = str(call.get("workdir") or workspace)
+        if not _within(root, workspace) or not _within(root, workdir):
+            return "WORKSPACE_MISMATCH"
+        return ""
+
+    def _epoch(self) -> int:
+        path = self.directory / "epoch.txt"
+        if not path.exists():
+            return 0
+        return int(path.read_text(encoding="utf-8").strip() or "0")
+
+    def _read_grants(self) -> list[dict[str, Any]]:
+        path = self.directory / "grants.json"
+        if not path.exists():
+            return []
+        payload = json.loads(path.read_text(encoding="utf-8"))
+        if not isinstance(payload, list):
+            return []
+        return [item for item in payload if isinstance(item, dict)]
+
+    def _write_grants(self, grants: list[dict[str, Any]]) -> None:
+        (self.directory / "grants.json").write_text(
+            json.dumps(grants, ensure_ascii=False),
+            encoding="utf-8",
+        )
+
+    def _audit(self, row: dict[str, Any]) -> None:
+        path = self.directory / "audit.jsonl"
+        with path.open("a", encoding="utf-8") as handle:
+            handle.write(json.dumps(row, ensure_ascii=False) + "\n")
+
+
+def confirm_from_local_ui(store: GrantStore, request: dict[str, Any]) -> dict[str, Any]:
+    """Trusted local confirm. A cloud permission string is not a confirm."""
+    return store.confirm(request)
+

+def install_grant_store(device_id: str, store: GrantStore) -> None:
+    _STORES[device_id] = store
+
+
+def clear_grant_store(device_id: str) -> None:
+    _STORES.pop(device_id, None)
+    _SENDERS.pop(device_id, None)
+
+
+def bind_click_sender(device_id: str, sender: Callable[[tuple[int, int]], None]) -> None:
+    _SENDERS[device_id] = sender
+
+
+def click_sender_for(device_id: str) -> Callable[[tuple[int, int]], None] | None:
+    return _SENDERS.get(device_id)
+
+
+def store_for(device_id: str) -> GrantStore:
+    existing = _STORES.get(device_id)
+    if existing is not None:
+        return existing
+    from RxyCode.RxyCode1_1_0.config.settings import get_data_dir
+
+    store = GrantStore(get_data_dir() / "device-grants" / device_id)
+    _STORES[device_id] = store
+    return store
+
+
+def _within(root: str, candidate: str) -> bool:
+    if not root or not candidate:
+        return False
+    base = Path(root).resolve()
+    target = Path(candidate).resolve()
+    try:
+        target.relative_to(base)
+    except ValueError:
+        return False
+    return True

+"""Cloud-side Remote Full requests. This mirror cannot confirm a local grant."""
+
+from __future__ import annotations
+
+from typing import Any
+
+
+class GrantMirror:
+    """What the browser is allowed to see while the runner keeps the real grant."""
+
+    def __init__(self) -> None:
+        self.rows: dict[str, dict[str, Any]] = {}
+
+    def request_remote_full(
+        self,
+        *,
+        device_id: str,
+        session_id: str,
+        run_id: str,
+        workspace: str,
+        capabilities: list[str],
+        expires_at: float,
+    ) -> dict[str, Any]:
+        row = {
+            "device_id": device_id,
+            "session_id": session_id,
+            "run_id": run_id,
+            "workspace": workspace,
+            "capabilities": list(capabilities),
+            "expires_at": expires_at,
+            "status": "waiting_for_local_confirmation",
+            "permission": None,
+            "grant_id": None,
+        }
+        self.rows[device_id] = row
+        return dict(row)
+
+    def forge_full(self, device_id: str) -> dict[str, Any]:
+        row = self.rows.setdefault(
+            device_id,
+            {
+                "device_id": device_id,
+                "status": "waiting_for_local_confirmation",
+                "permission": None,
+            },
+        )
+        row["permission"] = "full"
+        return dict(row)
+
+    def ack(self, device_id: str, *, grant_id: str, revoke_epoch: int) -> dict[str, Any]:
+        row = self.rows[device_id]
+        row["status"] = "granted"
+        row["grant_id"] = grant_id
+        row["revoke_epoch"] = revoke_epoch
+        row["permission"] = None
+        return dict(row)
+
+    def view(self, device_id: str) -> dict[str, Any]:
+        row = self.rows.get(device_id)
+        if row is None:
+            return {"device_id": device_id, "status": "waiting_for_local_confirmation"}
+        return dict(row)

+def _render(view: dict[str, object]) -> str:
+    script = REPO / "tests" / "web" / "w4" / "_render_c05.mjs"
+    payload = REPO / "tests" / "web" / "w4" / "_render_c05.json"
+    payload.write_text(json.dumps(view), encoding="utf-8")
+    component = REPO / "apps" / "web" / "src" / "features" / "devices" / "DeviceGrants.tsx"
+    script.write_text(
+        "\n".join(
+            [
+                'import { createElement } from "react";',
+                'import { renderToStaticMarkup } from "react-dom/server";',
+                'import { readFileSync } from "node:fs";',
+                f"import {{ DeviceGrants }} from {json.dumps(component.as_posix())};",
+                f"const view = JSON.parse(readFileSync({json.dumps(payload.as_posix())}, 'utf8'));",
+                "process.stdout.write(renderToStaticMarkup(createElement(DeviceGrants, { view })));",
+            ]
+        ),
+        encoding="utf-8",
+    )
+    ran = subprocess.run(
+        [str(BUN), str(script)],
+        cwd=WEB,
+        capture_output=True,
+        text=True,
+        timeout=30,
+        check=False,
+    )
+    script.unlink(missing_ok=True)
+    payload.unlink(missing_ok=True)
+    if ran.returncode != 0:
+        raise AssertionError(ran.stderr or ran.stdout)
+    return ran.stdout

+def _pair(gateway: object, device_id: str) -> threading.Thread:
+    from click.testing import CliRunner
+    from RxyCode.RxyCode1_1_0.main import cli
+    from rxycode_runner.remote.client import reset_stop
+
+    reset_stop()
+    browser = gateway.browser_session("tenant-a")  # type: ignore[attr-defined]
+    csrf = "csrf-token-value-for-c05"
+    issued = httpx.post(
+        f"{gateway.http}/api/v1/devices/pairings",  # type: ignore[attr-defined]
+        headers={"Origin": gateway.http, "X-CSRF-Token": csrf},
+        cookies={"rxy_session": browser, "rxy_csrf": csrf},
+        timeout=5,
+    )
+    issued.raise_for_status()
+    holder: dict[str, object] = {}
+
+    def launch() -> None:
+        holder["result"] = CliRunner().invoke(
+            cli,
+            [
+                "connect",
+                "--url",
+                gateway.ws,  # type: ignore[attr-defined]
+                "--code",
+                issued.json()["code"],
+                "--device-id",
+                device_id,
+                "--os-name",
+                "Windows",
+                "--capability",
+                "click",
+                "--heartbeat-seconds",
+                "0.2",
+            ],
+        )
+
+    thread = threading.Thread(target=launch)
+    thread.start()
+    deadline = time.time() + 10
+    while time.time() < deadline and not gateway.session_id(device_id):  # type: ignore[attr-defined]
+        time.sleep(0.05)
+    if not gateway.session_id(device_id):  # type: ignore[attr-defined]
+        raise AssertionError(holder.get("result"))
+    return thread

+def test_w4_c05_contract(tmp_path: Path) -> None:
+    """A local confirm lets shell and click in scope skip per-item approval."""
+    from uuid import uuid4
+
+    from RxyCode.RxyCode1_1_0.core.safety.approval import get_approval_broker
+    from rxycode_runner.local.grant_store import (
+        GrantStore,
+        bind_click_sender,
+        confirm_from_local_ui,
+        install_grant_store,
+    )
+    from rxycode_runner.remote.client import request_stop
+    from rxycode_runner.transport.tool_rpc import clear_local_tools
+    from rxycode_server.devices.gateway import serve_in_thread
+    from rxycode_server.devices.grants import GrantMirror
+
+    device_id = "desk-c05"
+    clock = {"t": 1_000.0}
+    work = REPO / "tests" / "web" / "w4" / "_bash_c05"
+    work.mkdir(parents=True, exist_ok=True)
+    marker = work / "grant-mark.txt"
+    script = work / "mark.py"
+    script.write_text(
+        "from pathlib import Path\n"
+        f"p = Path({str(marker)!r})\n"
+        "old = p.read_text(encoding='utf-8') if p.exists() else ''\n"
+        "p.write_text(old + 'x', encoding='utf-8')\n",
+        encoding="utf-8",
+    )
+    gateway = serve_in_thread(secret=_SECRET)
+    thread = _pair(gateway, device_id)
+    browser = gateway.browser_session("tenant-a")
+    csrf = "csrf-token-value-for-c05"
+    points: list[tuple[int, int]] = []
+    store = GrantStore(tmp_path / "grants", clock=lambda: clock["t"])
+    install_grant_store(device_id, store)
+    bind_click_sender(device_id, points.append)
+    mirror = GrantMirror()
+    session_id, run_id = "sess-c05", "run-c05"
+    try:
+        request = mirror.request_remote_full(
+            device_id=device_id,
+            session_id=session_id,
+            run_id=run_id,

+        assert store.confirmed_count() == 0

+        assert forged["permission"] == "full"
+        assert store.confirmed_count() == 0
+        assert "waiting_for_local_confirmation" in _render(mirror.view(device_id))
+        saved = confirm_from_local_ui(store, request)

+            device_id,

+        granted = _render(mirror.view(device_id))

+        assert "shell" in granted and "click" in granted and device_id in granted
+        broker = get_approval_broker()

+        def _count(event: dict[str, object]) -> None:

+

+            broker.set_event_sink(_count)
+        bash = _invoke(

+            browser,
+            csrf,
+            {
