+"""Tenant token, task, concurrency, preview, and sandbox quotas.
+
+The authenticated tenant and user are the only identity. Request headers are
+ignored. A refusal records a quota.exceeded event, does not occupy a worker,
+and does not call the model. Admitted runs add their cost to the durable
+run.created event so the next admission sees it.
+"""

+DEFAULT_LIMITS_TOKENS = 10_000_000
+DEFAULT_LIMITS_CONCURRENCY = 10_000
+DEFAULT_LIMITS_FEE = 100_000

+class WorkerFull(RuntimeError):
+    """This user already holds the concurrency ceiling."""

+class WorkerCapacity(Protocol):
+    """The scheduler side of admission. Only an admitted Run occupies a worker."""

+    workers: list[tuple[str, str, str]]
+
+    def acquire(self, tenant_id: str, user_id: str, *, limit: int, ttl_seconds: float = 3600) -> str:
+        """Occupy one worker, or refuse when the live count is at the ceiling."""

+    def release(self, tenant_id: str, user_id: str) -> None:

+class QuotaRefused(Exception):

+class RunWorkers:
+    """In-process worker slots used when the cloud scheduler is not injected."""
+
+    def __init__(self) -> None:
+        self.workers: list[tuple[str, str, str]] = []
+        self._deadlines: dict[str, float] = {}
+        self._timers: dict[str, threading.Timer] = {}
+        self._lock = threading.Lock()
+
+    @property
+    def live_ids(self) -> list[str]:
+        return []
+
+    def acquire(self, tenant_id: str, user_id: str, *, limit: int, ttl_seconds: float = 3600) -> str:
+        self._reap()
+        with self._lock:
+            held = _held(self.workers, tenant_id, user_id)
+            if limit < 1 or held >= limit:
+                raise WorkerFull(f"{tenant_id}/{user_id} is at concurrency {limit}")
+            worker_id = "worker-" + uuid.uuid4().hex[:8]
+            self.workers.append((tenant_id, user_id, worker_id))
+            self._deadlines[worker_id] = time.monotonic() + ttl_seconds
+        timer = threading.Timer(ttl_seconds, self._reap)
+        timer.daemon = True
+        with self._lock:
+            self._timers[worker_id] = timer
+        timer.start()
+        return worker_id
+
+    def release(self, tenant_id: str, user_id: str) -> None:
+        timer: threading.Timer | None = None
+        with self._lock:
+            for index in range(len(self.workers) - 1, -1, -1):
+                owner, user, worker_id = self.workers[index]
+                if owner == tenant_id and user == user_id:
+                    del self.workers[index]
+                    self._deadlines.pop(worker_id, None)
+                    timer = self._timers.pop(worker_id, None)
+                    break
+        if timer is not None:
+            timer.cancel()
+
+    def _reap(self) -> None:
+        now = time.monotonic()
+        expired: list[threading.Timer | None] = []
+        with self._lock:
+            kept: list[tuple[str, str, str]] = []
+            for owner, user, worker_id in self.workers:
+                if self._deadlines.get(worker_id, now + 1) <= now:
+                    expired.append(self._timers.pop(worker_id, None))
+                    self._deadlines.pop(worker_id, None)
+                else:
+                    kept.append((owner, user, worker_id))
+            self.workers[:] = kept
+        for timer in expired:
+            if timer is not None:
+                timer.cancel()
+
+
+def _held(workers: list[tuple[str, str, str]], tenant_id: str, user_id: str) -> int:
+    return sum(1 for owner, user, _worker in workers if owner == tenant_id and user == user_id)

+def _token_count(message: str) -> int:
+    return len([part for part in message.split() if part])

+class QuotaGate:
+    """Admit a Run only when this user still has credentials, budget, and quota."""
+
+    def __init__(
+        self,
+        *,
+        limits: Limits,
+        capacity: WorkerCapacity,
+        worker_ttl: float = 3600,
+    ) -> None:
+        self._limits = limits
+        self.capacity = capacity
+        self._worker_ttl = worker_ttl
+        self._events: dict[str, list[dict[str, object]]] = {}
+
+    def events_for(self, tenant_id: str) -> list[dict[str, object]]:
+        return list(self._events.get(tenant_id, []))
+
+    def precheck(
+        self,
+        *,
+        tenant_id: str,
+        user_id: str,
+        message: str,
+        headers: dict[str, str] | None,
+        credentials: str | None,
+        budget: int | None,
+        model: ModelClient | None,
+    ) -> QuotaDecision | None:
+        """Refuse before opening a database when the request itself cannot run."""
+        del headers
+        if model is not None and (credentials is None or credentials == ""):
+            return self._refuse(tenant_id, user_id, "credentials")
+        if model is not None and (budget is None or budget <= 0):
+            return self._refuse(tenant_id, user_id, "budget")
+        incoming = _token_count(message)
+        if incoming > self._limits.tokens:
+            return self._refuse(tenant_id, user_id, "token")
+        return None
+
+    def reserve(
+        self,
+        db: Session,
+        *,
+        tenant_id: str,
+        user_id: str,
+        message: str,
+    ) -> QuotaDecision | Reservation:
+        """Reserve one worker, or return the refusal. The Run row is not inserted here."""
+        usage = _usage(db, tenant_id)
+        incoming = _token_count(message)
+        limits = self._limits
+        if usage.tasks >= limits.tasks:
+            return self._refuse(tenant_id, user_id, "task")
+        if usage.tokens + incoming > limits.tokens:
+            return self._refuse(tenant_id, user_id, "token")
+        if usage.preview_fee + 1 > limits.preview_fee:
+            return self._refuse(tenant_id, user_id, "preview")
+        if usage.sandbox_fee + 1 > limits.sandbox_fee:
+            return self._refuse(tenant_id, user_id, "sandbox")
+        try:
+            self.capacity.acquire(
+                tenant_id,
+                user_id,
+                limit=limits.concurrency,
+                ttl_seconds=self._worker_ttl,
+            )
+        except Exception as exc:
+            if isinstance(exc, WorkerFull) or exc.__class__.__name__ == "CapacityFull":
+                return self._refuse(tenant_id, user_id, "concurrency")
+            raise
+        return Reservation(tenant_id=tenant_id, user_id=user_id, tokens=incoming)
+
+    def release(self, tenant_id: str, user_id: str) -> None:
+        self.capacity.release(tenant_id, user_id)
+
+    def _refuse(self, tenant_id: str, user_id: str, kind: str) -> QuotaDecision:
+        limit = {
+            "credentials": 0,
+            "budget": 0,
+            "task": self._limits.tasks,
+            "token": self._limits.tokens,
+            "concurrency": self._limits.concurrency,
+            "preview": self._limits.preview_fee,
+            "sandbox": self._limits.sandbox_fee,
+        }[kind]
+        event: dict[str, object] = {
+            "type": "quota.exceeded",
+            "kind": kind,
+            "tenant_id": tenant_id,
+            "user_id": user_id,
+            "limit": limit,
+        }
+        self._events.setdefault(tenant_id, []).append(event)
+        body: dict[str, object] = {
+            "error": "quota.exceeded",
+            "kind": kind,
+            "tenant_id": tenant_id,
+            "user_id": user_id,
+            "limit": limit,
+        }
+        return QuotaDecision(http_status=429, body=body, event=event)
+

+_installed: QuotaGate | None = None
+_default: QuotaGate | None = None
+

+def install_gate(gate: QuotaGate | None) -> None:
+    """Install the process gate. None restores the high default ceilings."""
+    global _installed
+    _installed = gate
+
+
+def active_gate() -> QuotaGate:
+    """Return the installed gate, or a process-wide gate with high ceilings."""
+    global _default
+    if _installed is not None:
+        return _installed
+    if _default is None:
+        _default = QuotaGate(limits=DEFAULT_LIMITS, capacity=RunWorkers())
+    return _default

+def _usage(db: Session, tenant_id: str) -> _Usage:
+    task_count = db.scalar(select(func.count()).select_from(Run).where(Run.tenant_id == tenant_id))
+    events = db.scalars(
+        select(RunEvent).where(RunEvent.tenant_id == tenant_id, RunEvent.type == "run.created")
+    ).all()
+    usage = _Usage(tasks=int(task_count or 0))
+    for event in events:
+        payload = event.payload if isinstance(event.payload, dict) else {}
+        quota = payload.get("quota") if isinstance(payload.get("quota"), dict) else {}
+        message = str(payload.get("message") or "")
+        usage.tokens += int(quota.get("tokens") or _token_count(message))
+        usage.preview_fee += int(quota.get("preview_fee") or 0)
+        usage.sandbox_fee += int(quota.get("sandbox_fee") or 0)
+    return usage

+"""Concurrency, cold-start samples, and sandbox GC.
+
+A failed or expired sandbox is removed with docker rm. Cold-start percentiles
+come from the measured create durations kept on this scheduler.
+"""
+
+from __future__ import annotations

+import base64
+import json
+import re
+import threading
+import time
+import uuid

+class TopologyRefused(ValueError):
+    """The compose file asks for privileged mode or omits a healthcheck."""
+
+
+class CapacityFull(RuntimeError):
+    """This user already holds every worker slot."""
+

+@dataclass(frozen=True)
+class ServiceSpec:
+    image: str
+    healthcheck: bool
+    privileged: bool
+
+
+@dataclass
+class _Record:
+    sandbox_id: str
+    tenant_id: str
+    deadline: float
+    failed: bool
+    backend: GVisorSandbox
+    timer: threading.Timer | None = None

+def load_compose(path: Path) -> dict[str, ServiceSpec]:
+    """Read the self-hosted topology. Privileged services are refused."""
+    text = Path(path).read_text(encoding="utf-8")
+    services = _parse_services(text)
+    if not services:
+        raise TopologyRefused("compose has no services")
+    for name, spec in services.items():
+        if spec.privileged:
+            raise TopologyRefused(f"{name} is privileged")
+        if not spec.healthcheck:
+            raise TopologyRefused(f"{name} has no healthcheck")
+        if not spec.image:
+            raise TopologyRefused(f"{name} has no image")
+    return services
+
+
+def _parse_services(text: str) -> dict[str, ServiceSpec]:
+    in_services = False
+    current: str | None = None
+    found: dict[str, dict[str, object]] = {}
+    for raw in text.splitlines():
+        stripped = raw.strip()
+        if not stripped or stripped.startswith("#"):
+            continue
+        indent = len(raw) - len(raw.lstrip(" "))
+        if indent == 0 and stripped == "services:":
+            in_services = True
+            current = None
+            continue
+        if not in_services:
+            continue
+        if indent == 0:
+            in_services = False
+            current = None
+            continue
+        if indent == 2 and stripped.endswith(":"):
+            current = stripped[:-1].strip()
+            found[current] = {"image": "", "healthcheck": False, "privileged": False}
+            continue
+        if current is None or indent != 4 or ":" not in stripped:
+            continue
+        key, value = stripped.split(":", 1)
+        slot = found[current]
+        if key.strip() == "image":
+            slot["image"] = value.strip().strip('"').strip("'")
+        elif key.strip() == "privileged":
+            slot["privileged"] = value.strip().lower() == "true"
+        elif key.strip() == "healthcheck":
+            slot["healthcheck"] = True
+    return {
+        name: ServiceSpec(
+            image=str(slot["image"]),
+            healthcheck=bool(slot["healthcheck"]),
+            privileged=bool(slot["privileged"]),
+        )
+        for name, slot in found.items()
+    }

+class Capacity:
+    """Worker occupancy plus the sandboxes this scheduler created."""
+
+    def __init__(self) -> None:
+        self.workers: list[tuple[str, str, str]] = []
+        self._deadlines: dict[str, float] = {}
+        self._worker_timers: dict[str, threading.Timer] = {}
+        self._records: dict[str, _Record] = {}
+        self._samples: list[float] = []
+        self._lock = threading.Lock()
+
+    @property
+    def live_ids(self) -> list[str]:
+        with self._lock:
+            return list(self._records)
+
+    def acquire(
+        self,
+        tenant_id: str,
+        user_id: str,
+        *,
+        limit: int,
+        ttl_seconds: float = 3600,
+    ) -> str:
+        """Occupy one worker until complete_run or the TTL, whichever comes first."""
+        self._reap_workers()
+        with self._lock:
+            held = sum(
+                1 for owner, user, _worker in self.workers if owner == tenant_id and user == user_id
+            )
+            if limit < 1 or held >= limit:
+                raise CapacityFull(f"{tenant_id}/{user_id} is at concurrency {limit}")
+            worker_id = "worker-" + uuid.uuid4().hex[:8]
+            self.workers.append((tenant_id, user_id, worker_id))
+            self._deadlines[worker_id] = time.monotonic() + ttl_seconds
+        timer = threading.Timer(ttl_seconds, self._reap_workers)
+        timer.daemon = True
+        with self._lock:
+            self._worker_timers[worker_id] = timer
+        timer.start()
+        return worker_id
+
+    def release(self, tenant_id: str, user_id: str) -> None:
+        timer: threading.Timer | None = None
+        with self._lock:
+            for index in range(len(self.workers) - 1, -1, -1):
+                owner, user, _worker = self.workers[index]
+                if owner == tenant_id and user == user_id:
+                    del self.workers[index]
+                    timer = self._worker_timers.pop(_worker, None)
+                    self._deadlines.pop(_worker, None)
+                    break
+        if timer is not None:
+            timer.cancel()
+
+    def _reap_workers(self) -> None:
+        now = time.monotonic()
+        expired: list[threading.Timer | None] = []
+        with self._lock:
+            kept: list[tuple[str, str, str]] = []
+            for owner, user, worker_id in self.workers:
+                if self._deadlines.get(worker_id, now + 1) <= now:
+                    expired.append(self._worker_timers.pop(worker_id, None))
+                    self._deadlines.pop(worker_id, None)
+                else:
+                    kept.append((owner, user, worker_id))
+            self.workers[:] = kept
+        for timer in expired:
+            if timer is not None:
+                timer.cancel()
+

+            workspace_id=workspace_id,

+            tenant_id=handle.tenant_id,
+            deadline=time.monotonic() + int(policy.ttl_seconds),
+            failed=False,

+        )

+        timer.daemon = True

+            self._samples.append(elapsed)
+            self._records[handle.sandbox_id] = record
+        timer.start()
+        return handle

+    @contextmanager
+    def sandbox(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        ttl_seconds: int | None = None,
+    ) -> Iterator[SandboxHandle]:
+        """Mark the sandbox failed when the caller raises after create."""
+        handle = self.provision(
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            ttl_seconds=ttl_seconds,
+        )
+        try:
+            yield handle
+        except Exception:
+            with self._lock:
+                record = self._records.get(handle.sandbox_id)
+                if record is not None:
+                    record.failed = True
+            self.gc()

+    def _expire(self, sandbox_id: str) -> None:
+        """TTL callback. gc() is the reaper; the test does not have to call it."""
+        with self._lock:
+            record = self._records.get(sandbox_id)
+            if record is not None:
+                record.deadline = min(record.deadline, time.monotonic())
+        self.gc()

+    def gc(self) -> list[str]:
+        """Remove failed sandboxes and sandboxes whose TTL has elapsed."""
+        now = time.monotonic()
+        with self._lock:
+            chosen = [
+                record
+                for record in self._records.values()
+                if record.failed or now >= record.deadline
+            ]
+        removed: list[str] = []
+        for record in chosen:
+            self._remove(record)
+            removed.append(record.sandbox_id)
+        return removed

+    def destroy_all(self) -> None:
+        with self._lock:
+            records = list(self._records.values())
+        for record in records:
+            self._remove(record)

+        with self._lock:

+        return self._percentile(50)

+        )

+    def read_events(

+        if not isinstance(payload, dict):

+        index = max(0, -(-len(ordered) * percentile // 100) - 1)

+def test_w5_c06_negative() -> None:
+    """GC removes a sandbox created before an exception and an expired one."""
+    from rxycode_worker.scheduler.capacity import Capacity
+
+    capacity = Capacity()
+    failed_id = ""
+    short_id = ""
+    healthy_id = ""
+    try:
+        healthy = capacity.provision(tenant_id="tenant-ok", workspace_id="ws-ok")
+        healthy_id = healthy.sandbox_id
+        try:
+            with capacity.sandbox(tenant_id="tenant-bad", workspace_id="ws-bad") as failed:
+                failed_id = failed.sandbox_id
+                assert _inspect(failed_id).returncode == 0
+                raise RuntimeError("after sandbox create")
+        except RuntimeError as exc:
+            assert str(exc) == "after sandbox create"
+        assert _inspect(failed_id).returncode != 0
+        short = capacity.provision(
+            tenant_id="tenant-ttl",
+            workspace_id="ws-ttl",
+            ttl_seconds=1,
+        )
+        short_id = short.sandbox_id
+        time.sleep(2.5)
+        assert _inspect(short_id).returncode != 0
+        alive = json.loads(_inspect(healthy_id).stdout)[0]
+        assert alive["State"]["Status"] == "running"
+        assert alive["Config"]["Labels"]["rxy.tenant"] == "tenant-ok"
+        samples = capacity.cold_start_samples()
+        assert len(samples) >= 3
+        assert all(sample > 0 for sample in samples)
+        assert capacity.cold_start_p50() == _nearest_rank(samples, 50)
+        assert capacity.cold_start_p95() == _nearest_rank(samples, 95)
+    finally:
+        capacity.destroy_all()
+        for sandbox_id in (failed_id, short_id, healthy_id):
+            if sandbox_id:
+                assert _inspect(sandbox_id).returncode != 0
+

+def _nearest_rank(samples: list[float], percentile: int) -> float:
+    ordered = sorted(samples)
+    index = max(0, -(-len(ordered) * percentile // 100) - 1)
+    return ordered[min(index, len(ordered) - 1)]
