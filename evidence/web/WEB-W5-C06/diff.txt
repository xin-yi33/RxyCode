+"""Tenant token, task, concurrency, preview, and sandbox quotas.
+
+The authenticated tenant and user are the only identity. Request headers are
+ignored. A refusal records a quota.exceeded event and does not occupy a worker
+or call the model.
+"""
+
+from __future__ import annotations
+
+from dataclasses import dataclass
+from typing import Protocol
+
+
+class WorkerCapacity(Protocol):
+    """The scheduler side of admission. Only an admitted Run occupies a worker."""
+
+    def acquire(self, tenant_id: str, user_id: str) -> str:
+        """Occupy one worker for this user."""
+
+    def release(self, tenant_id: str, user_id: str) -> None:
+        """Return the worker when admission cannot finish."""
+
+
+class ModelClient(Protocol):
+    """A model call. The gate invokes it only after admission succeeds."""
+
+    def __call__(self, prompt: str) -> str:
+        """Generate a reply for this prompt."""
+
+
+@dataclass(frozen=True)
+class Limits:
+    """Server-side ceilings for one user inside one tenant."""
+
+    tasks: int
+    tokens: int
+    concurrency: int
+    preview_fee: int
+    sandbox_fee: int
+
+
+@dataclass
+class _Usage:
+    tasks: int = 0
+    tokens: int = 0
+    concurrency: int = 0
+    preview_fee: int = 0
+    sandbox_fee: int = 0
+
+
+@dataclass(frozen=True)
+class QuotaDecision:
+    """Structured admission result. http_status 429 is a quota refusal."""
+
+    http_status: int
+    body: dict[str, object]
+    event: dict[str, object]
+
+
+class QuotaGate:
+    """Admit a Run only when this user still has credentials, budget, and quota."""
+
+    def __init__(self, *, limits: Limits, capacity: WorkerCapacity) -> None:
+        self._limits = limits
+        self.capacity = capacity
+        self._usage: dict[tuple[str, str], _Usage] = {}
+        self._events: dict[str, list[dict[str, object]]] = {}
+
+    def seed_usage(
+        self,
+        *,
+        tenant_id: str,
+        user_id: str,
+        tasks: int = 0,
+        tokens: int = 0,
+        concurrency: int = 0,
+        preview_fee: int = 0,
+        sandbox_fee: int = 0,
+    ) -> None:
+        """Record consumption that already happened on the server ledger."""
+        self._usage[(tenant_id, user_id)] = _Usage(
+            tasks=tasks,
+            tokens=tokens,
+            concurrency=concurrency,
+            preview_fee=preview_fee,
+            sandbox_fee=sandbox_fee,
+        )
+
+    def events_for(self, tenant_id: str) -> list[dict[str, object]]:
+        return list(self._events.get(tenant_id, []))
+
+    def create_run(
+        self,
+        *,
+        database_url: str,
+        tenant_id: str,
+        user_id: str,
+        session_id: str,
+        message: str,
+        target_id: str,
+        permission_profile: str,
+        request_id: str,
+        headers: dict[str, str] | None,
+        credentials: str | None,
+        budget: int | None,
+        model: ModelClient | None,
+    ) -> QuotaDecision:
+        """Refuse before any worker, model, or durable Run is created."""
+        del headers
+        usage = self._usage.get((tenant_id, user_id), _Usage())
+        kind = self._refusal_kind(usage, credentials=credentials, budget=budget)
+        if kind is not None:
+            return self._refuse(tenant_id, user_id, kind)
+        self.capacity.acquire(tenant_id, user_id)
+        try:
+            if model is not None:
+                model(message)
+            from rxycode_server.runs.service import create_run as persist_run
+
+            accepted = persist_run(
+                database_url,
+                tenant_id=tenant_id,
+                session_id=session_id,
+                message=message,
+                target_id=target_id,
+                permission_profile=permission_profile,
+                request_id=request_id,
+            )
+        except Exception:
+            self.capacity.release(tenant_id, user_id)
+            raise
+        return QuotaDecision(
+            http_status=accepted.http_status,
+            body={"status": accepted.status, "run_id": accepted.run_id},
+            event={"type": "run.accepted", "tenant_id": tenant_id, "user_id": user_id},
+        )
+
+    def _refusal_kind(
+        self,
+        usage: _Usage,
+        *,
+        credentials: str | None,
+        budget: int | None,
+    ) -> str | None:
+        if credentials is None or credentials == "":
+            return "credentials"
+        if budget is None or budget <= 0:
+            return "budget"
+        limits = self._limits
+        if usage.tasks >= limits.tasks:
+            return "task"
+        if usage.tokens >= limits.tokens:
+            return "token"
+        if usage.concurrency >= limits.concurrency:
+            return "concurrency"
+        if usage.preview_fee >= limits.preview_fee:
+            return "preview"
+        if usage.sandbox_fee >= limits.sandbox_fee:
+            return "sandbox"
+        return None
+
+    def _refuse(self, tenant_id: str, user_id: str, kind: str) -> QuotaDecision:
+        limit = {
+            "credentials": 0,
+            "budget": 0,
+            "task": self._limits.tasks,
+            "token": self._limits.tokens,
+            "concurrency": self._limits.concurrency,
+            "preview": self._limits.preview_fee,
+            "sandbox": self._limits.sandbox_fee,
+        }[kind]
+        event: dict[str, object] = {
+            "type": "quota.exceeded",
+            "kind": kind,
+            "tenant_id": tenant_id,
+            "user_id": user_id,
+            "limit": limit,
+        }
+        self._events.setdefault(tenant_id, []).append(event)
+        body: dict[str, object] = {
+            "error": "quota.exceeded",
+            "kind": kind,
+            "tenant_id": tenant_id,
+            "user_id": user_id,
+            "limit": limit,
+        }
+        return QuotaDecision(http_status=429, body=body, event=event)

+"""Concurrency, cold-start samples, and sandbox GC.
+
+A failed or expired sandbox is removed with docker rm. Cold-start percentiles
+come from the measured create durations kept on this scheduler.
+"""
+
+from __future__ import annotations
+
+import base64
+import json
+import re
+import threading
+import time
+import uuid
+from contextlib import contextmanager
+from dataclasses import dataclass, replace
+from pathlib import Path
+from typing import Iterator
+
+from rxycode_runner.sandbox.gvisor import GVisorSandbox
+from rxycode_runner.sandbox.policy import load_policy, reject_if_unsafe
+from rxycode_runner.sandbox.provider import ExecResult, SandboxHandle
+
+_NONCE = re.compile(r"^[0-9a-f]{16}$")
+
+
+class TopologyRefused(ValueError):
+    """The compose file asks for privileged mode or omits a healthcheck."""
+
+
+@dataclass(frozen=True)
+class ServiceSpec:
+    image: str
+    healthcheck: bool
+    privileged: bool
+
+
+@dataclass
+class _Record:
+    sandbox_id: str
+    tenant_id: str
+    deadline: float
+    failed: bool
+    backend: GVisorSandbox
+
+
+def load_compose(path: Path) -> dict[str, ServiceSpec]:
+    """Read the self-hosted topology. Privileged services are refused."""
+    text = Path(path).read_text(encoding="utf-8")
+    services = _parse_services(text)
+    if not services:
+        raise TopologyRefused("compose has no services")
+    for name, spec in services.items():
+        if spec.privileged:
+            raise TopologyRefused(f"{name} is privileged")
+        if not spec.healthcheck:
+            raise TopologyRefused(f"{name} has no healthcheck")
+        if not spec.image:
+            raise TopologyRefused(f"{name} has no image")
+    return services
+
+
+def _parse_services(text: str) -> dict[str, ServiceSpec]:
+    in_services = False
+    current: str | None = None
+    found: dict[str, dict[str, object]] = {}
+    for raw in text.splitlines():
+        stripped = raw.strip()
+        if not stripped or stripped.startswith("#"):
+            continue
+        indent = len(raw) - len(raw.lstrip(" "))
+        if indent == 0 and stripped == "services:":
+            in_services = True
+            current = None
+            continue
+        if not in_services:
+            continue
+        if indent == 0:
+            in_services = False
+            current = None
+            continue
+        if indent == 2 and stripped.endswith(":"):
+            current = stripped[:-1].strip()
+            found[current] = {"image": "", "healthcheck": False, "privileged": False}
+            continue
+        if current is None or indent != 4 or ":" not in stripped:
+            continue
+        key, value = stripped.split(":", 1)
+        slot = found[current]
+        if key.strip() == "image":
+            slot["image"] = value.strip().strip('"').strip("'")
+        elif key.strip() == "privileged":
+            slot["privileged"] = value.strip().lower() == "true"
+        elif key.strip() == "healthcheck":
+            slot["healthcheck"] = True
+    return {
+        name: ServiceSpec(
+            image=str(slot["image"]),
+            healthcheck=bool(slot["healthcheck"]),
+            privileged=bool(slot["privileged"]),
+        )
+        for name, slot in found.items()
+    }
+

+class Capacity:
+    """Worker occupancy plus the sandboxes this scheduler created."""
+
+    def __init__(self) -> None:
+        self.workers: list[tuple[str, str, str]] = []
+        self._records: dict[str, _Record] = {}
+        self._samples: list[float] = []
+        self._lock = threading.Lock()
+
+    @property
+    def live_ids(self) -> list[str]:
+        with self._lock:
+            return list(self._records)
+
+    def acquire(self, tenant_id: str, user_id: str) -> str:
+        """Occupy one worker. The quota gate calls this only after admission."""
+        worker_id = "worker-" + uuid.uuid4().hex[:8]
+        with self._lock:
+            self.workers.append((tenant_id, user_id, worker_id))
+        return worker_id
+
+    def release(self, tenant_id: str, user_id: str) -> None:
+        with self._lock:
+            for index in range(len(self.workers) - 1, -1, -1):
+                owner, user, _worker = self.workers[index]
+                if owner == tenant_id and user == user_id:
+                    del self.workers[index]
+                    return
+
+    def provision(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        ttl_seconds: int | None = None,
+    ) -> SandboxHandle:
+        """Create one isolated sandbox and record the measured cold start."""
+        started = time.perf_counter()
+        policy = load_policy()
+        if ttl_seconds is not None:
+            policy = replace(policy, ttl_seconds=ttl_seconds)
+            reject_if_unsafe(policy)
+        backend = GVisorSandbox()
+        handle = backend.create(
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            retain=True,
+            policy=policy,
+        )
+        elapsed = time.perf_counter() - started
+        record = _Record(
+            sandbox_id=handle.sandbox_id,
+            tenant_id=handle.tenant_id,
+            deadline=time.monotonic() + int(policy.ttl_seconds),
+            failed=False,
+            backend=backend,
+        )
+        with self._lock:
+            self._samples.append(elapsed)
+            self._records[handle.sandbox_id] = record
+        return handle
+
+    @contextmanager
+    def sandbox(
+        self,
+        *,
+        tenant_id: str,
+        workspace_id: str,
+        ttl_seconds: int | None = None,
+    ) -> Iterator[SandboxHandle]:
+        """Mark the sandbox failed when the caller raises after create."""
+        handle = self.provision(
+            tenant_id=tenant_id,
+            workspace_id=workspace_id,
+            ttl_seconds=ttl_seconds,
+        )
+        try:
+            yield handle
+        except Exception:
+            with self._lock:
+                record = self._records.get(handle.sandbox_id)
+                if record is not None:
+                    record.failed = True
+            raise
+
+    def gc(self) -> list[str]:
+        """Remove failed sandboxes and sandboxes whose TTL has elapsed."""
+        now = time.monotonic()
+        with self._lock:
+            chosen = [
+                record
+                for record in self._records.values()
+                if record.failed or now >= record.deadline
+            ]
+        removed: list[str] = []
+        for record in chosen:
+            self._remove(record)
+            removed.append(record.sandbox_id)
+        return removed
+
+    def destroy_all(self) -> None:
+        with self._lock:
+            records = list(self._records.values())
+        for record in records:
+            self._remove(record)
+
+    def cold_start_samples(self) -> list[float]:
+        with self._lock:
+            return list(self._samples)
+
+    def cold_start_p50(self) -> float:
+        return self._percentile(50)
+
+    def cold_start_p95(self) -> float:
+        return self._percentile(95)
+
+    def write_marker(self, handle: SandboxHandle, nonce: str) -> None:
+        if not _NONCE.fullmatch(nonce):
+            raise ValueError("marker nonce is refused")
+        code = (
+            "import pathlib; "
+            f"pathlib.Path('/tmp/tenant-marker').write_text('{nonce}')"
+        )
+        self._exec(handle, handle.tenant_id, code)
+
+    def read_marker(self, handle: SandboxHandle, tenant_id: str | None = None) -> str:
+        owner = tenant_id or handle.tenant_id
+        result = self._exec(handle, owner, "print(open('/tmp/tenant-marker').read())")
+        return result.stdout.strip()
+
+    def publish_event(self, handle: SandboxHandle, event: dict[str, object]) -> None:
+        blob = base64.b64encode(json.dumps(event).encode()).decode()
+        code = (
+            "import base64, pathlib; "
+            f"pathlib.Path('/tmp/events.json').write_bytes(base64.b64decode('{blob}'))"
+        )
+        self._exec(handle, handle.tenant_id, code)
+
+    def read_events(
+        self,
+        handle: SandboxHandle,
+        tenant_id: str | None = None,
+    ) -> dict[str, object]:
+        owner = tenant_id or handle.tenant_id
+        result = self._exec(handle, owner, "print(open('/tmp/events.json').read())")
+        payload = json.loads(result.stdout)
+        if not isinstance(payload, dict):
+            raise RuntimeError("sandbox event payload is not an object")
+        return payload
+
+    def _percentile(self, percentile: int) -> float:
+        with self._lock:
+            ordered = sorted(self._samples)
+        if not ordered:
+            raise RuntimeError("no cold start samples")
+        index = max(0, -(-len(ordered) * percentile // 100) - 1)
+        return ordered[min(index, len(ordered) - 1)]
+
+    def _exec(self, handle: SandboxHandle, tenant_id: str, code: str) -> ExecResult:
+        with self._lock:
+            record = self._records[handle.sandbox_id]
+            backend = record.backend
+        result = backend.exec(
+            handle.sandbox_id,
+            tenant_id=tenant_id,
+            argv=["python", "-c", code],
+            timeout=30,
+        )
+        if result.exit_code != 0:
+            detail = result.stderr or result.stdout or "sandbox exec failed"
+            raise RuntimeError(detail)
+        return result
+
+    def _remove(self, record: _Record) -> None:
+        try:
+            record.backend.destroy(record.sandbox_id, tenant_id=record.tenant_id)
+        except Exception:
+            record.backend._docker(["rm", "-f", record.sandbox_id], timeout=30)
+        with self._lock:
+            self._records.pop(record.sandbox_id, None)

+import json
+import subprocess
+import sys
+import threading
+from pathlib import Path

+REPO = Path(__file__).resolve().parents[3]
+for _entry in (

+def _docker(*args: str) -> subprocess.CompletedProcess[str]:
+    from rxycode_runner.sandbox.gvisor import docker_executable
+
+    return subprocess.run(
+        [docker_executable(), *args],
+        capture_output=True,
+        text=True,
+        timeout=30,
+        check=False,
+    )

+def test_w5_c06_integration() -> None:
+    """Tenants A and B keep isolated workspace files, labels, and events."""
+    from rxycode_runner.sandbox.provider import AttachRefused
+    from rxycode_worker.scheduler.capacity import Capacity

+    capacity = Capacity()
+    nonce_a = "a1b2c3d4e5f60718"
+    nonce_b = "b9c8d7e6f5041320"
+    results: dict[str, object] = {}
+    errors: list[BaseException] = []
+    ready = threading.Barrier(2)
+
+    def work(tenant: str, workspace: str, nonce: str) -> None:
+        try:
+            handle = capacity.provision(tenant_id=tenant, workspace_id=workspace)
+            capacity.write_marker(handle, nonce)
+            capacity.publish_event(
+                handle,
+                {"type": "workspace.ready", "tenant_id": tenant, "nonce": nonce},
+            )

+            results[tenant] = handle
+        except BaseException as exc:
+            errors.append(exc)
+            try:
+                ready.abort()
+            except threading.BrokenBarrierError:
+                pass
+
+    threads = [
+        threading.Thread(target=work, args=("tenant-a", "ws-tenant-a", nonce_a)),
+        threading.Thread(target=work, args=("tenant-b", "ws-tenant-b", nonce_b)),
+    ]
+    try:
+        for thread in threads:
+            thread.start()
+        for thread in threads:
+            thread.join(timeout=120)

+        assert set(results) == {"tenant-a", "tenant-b"}
+        handle_a = results["tenant-a"]
+        handle_b = results["tenant-b"]

+        assert handle_a.workspace_id != handle_b.workspace_id

+        assert capacity.read_events(handle_a)["nonce"] == nonce_a

+        try:
+            capacity.read_marker(handle_b, tenant_id="tenant-a")
+        except AttachRefused:
+            pass
+        else:
+            raise AssertionError("tenant A read tenant B marker")
+        for handle, tenant, nonce, other in (

+def test_w5_c06_negative() -> None:
+    """GC removes a sandbox created before an exception and an expired one."""
+    from rxycode_worker.scheduler.capacity import Capacity
+
+    capacity = Capacity()
+    failed_id = ""
+    short_id = ""
+    healthy_id = ""
+    try:
+        healthy = capacity.provision(tenant_id="tenant-ok", workspace_id="ws-ok")
+        healthy_id = healthy.sandbox_id
+        try:
+            with capacity.sandbox(tenant_id="tenant-bad", workspace_id="ws-bad") as failed:
+                failed_id = failed.sandbox_id
+                assert _inspect(failed_id).returncode == 0
+                raise RuntimeError("after sandbox create")
+        except RuntimeError as exc:
+            assert str(exc) == "after sandbox create"
+        short = capacity.provision(
+            tenant_id="tenant-ttl",
+            workspace_id="ws-ttl",
+            ttl_seconds=1,
+        )
+        short_id = short.sandbox_id
+        time.sleep(1.5)
+        removed = set(capacity.gc())
+        assert failed_id in removed
+        assert short_id in removed
+        assert healthy_id not in removed
+        assert _inspect(failed_id).returncode != 0
+        assert _inspect(short_id).returncode != 0
+        alive = json.loads(_inspect(healthy_id).stdout)[0]
+        assert alive["State"]["Status"] == "running"
+        assert alive["Config"]["Labels"]["rxy.tenant"] == "tenant-ok"
+        samples = capacity.cold_start_samples()
+        assert len(samples) >= 3
+        assert all(sample > 0 for sample in samples)
+        assert capacity.cold_start_p50() == _nearest_rank(samples, 50)
+        assert capacity.cold_start_p95() == _nearest_rank(samples, 95)
+    finally:
+        capacity.destroy_all()
+        for sandbox_id in (failed_id, short_id, healthy_id):
+            if sandbox_id:
+                assert _inspect(sandbox_id).returncode != 0
+
+
+def _nearest_rank(samples: list[float], percentile: int) -> float:
+    ordered = sorted(samples)
+    index = max(0, -(-len(ordered) * percentile // 100) - 1)
+    return ordered[min(index, len(ordered) - 1)]
