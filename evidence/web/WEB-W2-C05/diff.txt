+class CheckpointerUnavailable(RuntimeError):

+class _ReadState(TypedDict, total=False):

+def graph_thread_id(session_id: str, run_id: str) -> str:
+    """Stable id. Length prefixes stop `a:b` + `c` colliding with `a` + `b:c`."""
+    if not isinstance(session_id, str) or not session_id or "\x00" in session_id:
+        raise ValueError("invalid session_id")
+    if not isinstance(run_id, str) or not run_id or "\x00" in run_id:
+        raise ValueError("invalid run_id")
+    return f"rxy:{len(session_id)}:{session_id}:{len(run_id)}:{run_id}"

+def _conninfo(database_url: str) -> str:
+    if not isinstance(database_url, str) or not database_url:
+        raise CheckpointerUnavailable("database url is missing")
+    url = database_url.replace("postgresql+psycopg://", "postgresql://", 1)
+    url = url.replace("postgres+psycopg://", "postgresql://", 1)
+    parts = urlparse(url)
+    if parts.scheme not in {"postgresql", "postgres"}:
+        raise CheckpointerUnavailable("checkpointer url is not postgres")
+    query = dict(parse_qsl(parts.query, keep_blank_values=True))
+    query.setdefault("connect_timeout", "3")
+    return urlunparse(parts._replace(query=urlencode(query)))

+def _read_node(state: _ReadState) -> dict[str, str]:
+    return {"observed": state["source"]}
+

+def _side_node(state: _ReadState) -> dict[str, str]:
+    """A later write. interrupt_before keeps a pure-read crash from reaching it."""
+    marker = state.get("marker") or ""
+    if marker:

+        Path(marker).write_text("side-effect", encoding="utf-8")

+def _compile(saver: object):
+    graph = StateGraph(_ReadState)
+    graph.add_node("read", _read_node)
+    graph.add_node("side", _side_node)
+    graph.add_edge(START, "read")
+    graph.add_edge("read", "side")
+    graph.add_edge("side", END)
+    return graph.compile(checkpointer=saver, interrupt_before=["side"])

+        raise CheckpointerUnavailable("AsyncPostgresSaver is not installed") from exc

+    entered = False
+    try:
+        saver = await saver_cm.__aenter__()
+        entered = True
+        await saver.setup()
+    except CheckpointerUnavailable:
+        if entered:
+            await saver_cm.__aexit__(*sys.exc_info())
+        raise
+    except Exception as exc:
+        if entered:
+            await saver_cm.__aexit__(type(exc), exc, exc.__traceback__)
+        raise CheckpointerUnavailable(f"checkpointer is unavailable: {exc}") from exc
+    try:
+        yield saver
+    finally:

+def _run_on_selector(factory: Callable[[], Awaitable[dict[str, object]]]) -> dict[str, object]:
+    """psycopg async refuses the Windows ProactorEventLoop, so use a selector loop."""
+    if sys.platform == "win32":
+        loop = asyncio.SelectorEventLoop(selectors.SelectSelector())
+    else:
+        loop = asyncio.new_event_loop()
+    asyncio.set_event_loop(loop)
+    try:
+        return loop.run_until_complete(factory())
+    finally:
+        loop.run_until_complete(loop.shutdown_asyncgens())
+        loop.close()
+        asyncio.set_event_loop(None)

+def _child_env() -> dict[str, str]:
+    env = os.environ.copy()
+    previous = env.get("PYTHONPATH", "")
+    parts = [str(REPO), str(_SERVER), str(_WORKER)]
+    if previous:
+        parts.append(previous)
+    env["PYTHONPATH"] = os.pathsep.join(parts)
+    return env

+def _resume_in_new_process(database_url: str, session_id: str, run_id: str) -> dict:
+    """Load the checkpoint from a second interpreter. That is the worker restart."""
+    script = (
+        "import asyncio, json, os\n"
+        "from rxycode_worker.checkpoints import resume_readonly\n"
+        "async def main():\n"
+        "    restored = await resume_readonly(\n"
+        "        os.environ['C05_URL'],\n"
+        "        session_id=os.environ['C05_SESSION'],\n"
+        "        run_id=os.environ['C05_RUN'],\n"
+        "    )\n"
+        "    print('C05_RESUME ' + json.dumps(restored))\n"
+        "asyncio.run(main())\n"
+    )
+    env = _child_env()
+    env["C05_URL"] = database_url
+    env["C05_SESSION"] = session_id
+    env["C05_RUN"] = run_id
+    proc = subprocess.run(
+        [sys.executable, "-c", script],
+        env=env,
+        capture_output=True,
+        text=True,
+        timeout=90,
+        check=False,
+    )
+    assert proc.returncode == 0, proc.stderr or proc.stdout
+    line = next(
+        item for item in proc.stdout.splitlines() if item.startswith("C05_RESUME ")
+    )
+    return json.loads(line.removeprefix("C05_RESUME "))

+@pytest.mark.asyncio
+async def test_w2_c05_contract(postgres_url: str, tmp_path: Path):
+    """A pure-read node survives a new process, and a second run does not share it."""
+    session_id = "session-c05"
+    run_a = "run-read-a"
+    run_b = "run-read-b"
+    marker = tmp_path / "side-effect.txt"
+    await record_readonly_checkpoint(
+        postgres_url,
+        session_id=session_id,
+        run_id=run_a,
+        source="alpha-snapshot",
+        marker=str(marker),
+    )
+    await record_readonly_checkpoint(
+        postgres_url,
+        session_id=session_id,
+        run_id=run_b,
+        source="beta-snapshot",
+        marker=str(marker),
+    )
+    assert not marker.exists()
+    restored = _resume_in_new_process(postgres_url, session_id, run_a)
+    other = _resume_in_new_process(postgres_url, session_id, run_b)
+    assert restored["observed"] == "alpha-snapshot"
+    assert other["observed"] == "beta-snapshot"
+    assert restored["thread_id"] == graph_thread_id(session_id, run_a)
+    assert other["thread_id"] == graph_thread_id(session_id, run_b)
+    assert restored["thread_id"] != other["thread_id"]
+    assert "side" in restored["next"]
+    assert not marker.exists()

+@pytest.mark.asyncio
+async def test_w2_c05_negative(tmp_path: Path):
+    """An unavailable checkpointer does not start an unjournaled side effect."""
+    marker = tmp_path / "unjournaled.txt"

+    assert outcome["started"] is False
+    assert outcome["status"] == "recovery_required"
+    assert not marker.exists()

+def test_w2_c05_corrupt_journal(tmp_path: Path, postgres_url: str):
+    """A damaged journal blocks the shell instead of guessing the outcome."""
+    attempt_id = "att_" + "ab" * 16
+    call_id = "call_" + "cd" * 16
+    journal = ToolExecutionJournal(tmp_path / "journal")
+    (journal.directory / f"{attempt_id}.json").write_text("{not-json", encoding="utf-8")
+    marker = tmp_path / "corrupt-write.txt"
+    command = [
+        "pwsh",
+        "-NoProfile",
+        "-Command",
+        f"Set-Content -LiteralPath '{marker}' -Value 'should-not-run'",
+    ]
+    outcome = recover_unknown_shell(
+        postgres_url,
+        tenant_id="tenant-c05",
+        run_id="run-missing",
+        call_id=call_id,
+        attempt_id=attempt_id,
+        command=command,
+        journal=journal,
+    )
+    assert outcome["executed"] is False
+    assert outcome["journal_state"] == "unknown"
+    assert outcome["run_status"] == "recovery_required"
+    assert not marker.exists()

+def test_w2_c05_remote_journal_states(tmp_path: Path):
+    """Remote call ids use not_started, unknown, and completed without a second write."""
+    journal = ToolExecutionJournal(tmp_path / "journal")
+    adapter = RemoteJournalAdapter(journal)
+    attempt_id = "att_" + "11" * 16
+    call_id = "call_" + "22" * 16
+    other_call = "call_" + "33" * 16
+    args = {"cmd": "echo hi"}
+    assert adapter.state(attempt_id=attempt_id, call_id=call_id) == "not_started"
+    first = adapter.reserve(
+        attempt_id=attempt_id,
+        call_id=call_id,
+        tool="shell",
+        args=args,
+    )
+    assert first.action == "execute"
+    assert adapter.state(attempt_id=attempt_id, call_id=call_id) == "unknown"
+    blocked = adapter.reserve(
+        attempt_id=attempt_id,
+        call_id=other_call,
+        tool="shell",
+        args=args,
+    )
+    assert blocked.action == "uncertain"
+    adapter.complete(
+        attempt_id=attempt_id,
+        call_id=call_id,
+        tool="shell",
+        args=args,
+        result="ok",
+    )
+    assert adapter.state(attempt_id=attempt_id, call_id=call_id) == "completed"
+    reused = adapter.reserve(
+        attempt_id=attempt_id,
+        call_id=call_id,
+        tool="shell",
+        args=args,
+    )
+    assert reused.action == "reuse"
+    assert reused.result == "ok"
