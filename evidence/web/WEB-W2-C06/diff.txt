+"""Durable approval requests. A decision is one CAS update and never runs a tool."""
+
+from __future__ import annotations
+
+import uuid
+from datetime import datetime, timezone
+from typing import Any
+
+from sqlalchemy import func, select
+from sqlalchemy import create_engine
+from sqlalchemy.orm import Session
+from sqlalchemy.pool import NullPool
+
+from rxycode_server.persistence.models import Approval, Outbox, Run, RunEvent
+
+
+def _engine(database_url: str):
+    return create_engine(database_url, poolclass=NullPool)
+
+
+class ApprovalStale(Exception):
+    """The approval was already decided, cancelled, or past its deadline."""
+
+    def __init__(self, approval_id: str) -> None:
+        super().__init__(approval_id)
+        self.approval_id = approval_id
+        self.http_status = 409
+        self.code = "APPROVAL_STALE"
+
+
+def _aware(value: datetime | None) -> datetime | None:
+    if value is not None and value.tzinfo is None:
+        return value.replace(tzinfo=timezone.utc)
+    return value
+
+
+def _now(value: datetime | None) -> datetime:
+    if value is None:
+        return datetime.now(timezone.utc)
+    current = _aware(value)
+    assert current is not None
+    return current
+
+
+def _lease_is_current(run: Run, worker_id: str, epoch: int, now: datetime) -> bool:
+    if run.lease_owner != worker_id or int(run.lease_epoch) != int(epoch):
+        return False
+    if run.status not in {"leased", "running", "cancel_requested"}:
+        return False
+    until = _aware(run.lease_until)
+    return until is not None and until >= now
+
+
+def _append(db: Session, run: Run, event_type: str, payload: dict[str, Any]) -> None:
+    retained = db.scalar(select(func.max(RunEvent.seq)).where(RunEvent.run_id == run.id))
+    seq = max(int(run.event_seq or 0), int(retained or 0)) + 1
+    run.event_seq = seq
+    event_id = uuid.uuid4().hex
+    db.add(
+        RunEvent(
+            event_id=event_id,
+            run_id=run.id,
+            tenant_id=run.tenant_id,
+            seq=seq,
+            type=event_type,
+            payload=payload,
+            at=datetime.now(timezone.utc),
+        )
+    )
+    db.flush()
+    db.add(Outbox(id=uuid.uuid4().hex, event_id=event_id, dispatch_status="pending", attempts=0))
+
+
+def _lock_run(db: Session, tenant_id: str, run_id: str) -> Run:
+    return db.execute(
+        select(Run).where(Run.id == run_id, Run.tenant_id == tenant_id).with_for_update()
+    ).scalar_one()
+

+def request_approval(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    call_id: str,
+    tool: str,
+    permission_scope: str,
+    arguments: dict[str, Any],
+    deadline: datetime,
+    worker_id: str,
+    epoch: int,
+    attempt_id: str | None = None,
+) -> dict[str, Any]:
+    """Persist the ask, emit approval.requested, and release the worker lease."""
+    engine = _engine(database_url)
+    approval_id = uuid.uuid4().hex
+    stored_attempt = attempt_id or f"att_{uuid.uuid4().hex}"
+    try:
+        with Session(engine) as db:
+            run = _lock_run(db, tenant_id, run_id)
+            now = datetime.now(timezone.utc)
+            if not _lease_is_current(run, worker_id, epoch, now):
+                raise ApprovalStale(approval_id)
+            db.add(
+                Approval(
+                    id=approval_id,
+                    tenant_id=tenant_id,
+                    run_id=run_id,
+                    call_id=call_id,
+                    status="pending",
+                    revision=0,
+                    deadline=_aware(deadline),
+                )
+            )
+            db.flush()
+            _append(
+                db,
+                run,
+                "approval.requested",
+                {
+                    "approval_id": approval_id,
+                    "call_id": call_id,
+                    "tool": tool,
+                    "permission_scope": permission_scope,
+                    "arguments": arguments,
+                    "attempt_id": stored_attempt,
+                    "revision": 0,
+                },
+            )
+            run.status = "waiting_approval"
+            run.lease_owner = None
+            run.lease_until = None
+            run.revision = int(run.revision or 0) + 1
+            db.commit()
+    finally:
+        engine.dispose()
+    return {
+        "approval_id": approval_id,
+        "status": "pending",
+        "revision": 0,
+        "attempt_id": stored_attempt,
+    }
+
+
+def _requested_payload(db: Session, approval_id: str) -> dict[str, Any]:
+    events = db.scalars(
+        select(RunEvent).where(RunEvent.type == "approval.requested").order_by(RunEvent.seq)
+    ).all()
+    for event in events:
+        payload = event.payload or {}
+        if payload.get("approval_id") == approval_id:
+            return dict(payload)
+    return {}
+

+def list_pending(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    now: datetime | None = None,
+) -> list[dict[str, Any]]:
+    """Return asks that are still pending at `now`, including the original scope."""
+    current = _now(now)
+    engine = _engine(database_url)
+    try:
+        with Session(engine) as db:
+            rows = db.scalars(
+                select(Approval).where(
+                    Approval.tenant_id == tenant_id,
+                    Approval.run_id == run_id,
+                    Approval.status == "pending",
+                )
+            ).all()
+            found: list[dict[str, Any]] = []
+            for row in rows:
+                deadline = _aware(row.deadline)
+                if deadline is not None and deadline <= current:
+                    continue
+                payload = _requested_payload(db, row.id)
+                found.append(
+                    {
+                        "approval_id": row.id,
+                        "call_id": row.call_id,
+                        "tool": payload.get("tool", ""),
+                        "permission_scope": payload.get("permission_scope", ""),
+                        "arguments": payload.get("arguments", {}),
+                        "attempt_id": payload.get("attempt_id", ""),
+                        "status": row.status,
+                        "revision": int(row.revision or 0),
+                        "deadline": deadline.isoformat() if deadline is not None else None,
+                    }
+                )
+            return found
+    finally:
+        engine.dispose()
+

+def get_approval(database_url: str, *, tenant_id: str, approval_id: str) -> dict[str, Any] | None:
+    engine = _engine(database_url)
+    try:
+        with Session(engine) as db:
+            row = db.get(Approval, approval_id)
+            if row is None or row.tenant_id != tenant_id:
+                return None
+            payload = _requested_payload(db, row.id)
+            return {
+                "approval_id": row.id,
+                "run_id": row.run_id,
+                "call_id": row.call_id,
+                "status": row.status,
+                "revision": int(row.revision or 0),
+                "attempt_id": str(payload.get("attempt_id") or ""),
+                "tool": payload.get("tool", ""),
+                "permission_scope": payload.get("permission_scope", ""),
+            }
+    finally:
+        engine.dispose()
+
+
+def _reject_if_closed(row: Approval, expected_revision: int, now: datetime) -> None:
+    if int(row.revision or 0) != int(expected_revision) or row.status != "pending":
+        raise ApprovalStale(row.id)
+    deadline = _aware(row.deadline)
+    if deadline is not None and deadline <= now:
+        row.status = "expired"
+        row.revision = int(row.revision or 0) + 1
+        raise ApprovalStale(row.id)
+

+def decide_approval(
+    database_url: str,
+    *,
+    tenant_id: str,
+    approval_id: str,
+    decision: str,
+    expected_revision: int,
+    now: datetime | None = None,
+) -> dict[str, Any]:
+    """Record one decision. This function does not execute a tool."""
+    if decision not in {"approved", "rejected"}:
+        raise ApprovalStale(approval_id)
+    engine = _engine(database_url)
+    try:
+        with Session(engine) as db:
+            row = db.execute(
+                select(Approval)
+                .where(Approval.id == approval_id, Approval.tenant_id == tenant_id)
+                .with_for_update()
+            ).scalar_one()
+            current = datetime.now(timezone.utc) if now is None else _now(now)
+            try:
+                _reject_if_closed(row, expected_revision, current)
+            except ApprovalStale:
+                db.commit()
+                raise
+            row.status = decision
+            row.revision = int(row.revision or 0) + 1
+            run = _lock_run(db, tenant_id, row.run_id)
+            _append(
+                db,
+                run,
+                "approval.resolved",
+                {
+                    "approval_id": row.id,
+                    "call_id": row.call_id,
+                    "decision": decision,
+                    "revision": int(row.revision),
+                },
+            )
+            run.lease_owner = None
+            run.lease_until = None
+            run.status = "pending" if decision == "approved" else "cancelled"
+            run.revision = int(run.revision or 0) + 1
+            revision = int(row.revision)
+            db.commit()
+    finally:
+        engine.dispose()
+    return {"status": decision, "revision": revision, "http_status": 200}
+
+
+def cancel_approval(

+                run,
+                "approval.resolved",

+                    "approval_id": row.id,
+                    "call_id": row.call_id,

+                    "revision": int(row.revision),

+            run.status = "cancelled"
+            run.lease_owner = None
+            run.lease_until = None
+            run.revision = int(run.revision or 0) + 1

+    finally:

+    return {"status": "cancelled", "http_status": 200}

+"""Pause a run on a LangGraph interrupt. The tool runs only after resume."""
+
+from __future__ import annotations
+
+import asyncio
+import threading
+import uuid
+from datetime import datetime
+from typing import Any, TypedDict
+
+from langgraph.graph import END, START, StateGraph
+from langgraph.types import Command, interrupt
+
+from rxycode_server.approvals.service import get_approval, request_approval
+from rxycode_server.runs.service import load_run
+from rxycode_worker.checkpoints import _call_on_selector, _open_saver, graph_thread_id
+from rxycode_worker.supervisor import claim
+
+
+class _GateState(TypedDict, total=False):
+    approval_id: str
+    attempt_id: str
+    call_id: str
+    run_id: str
+    journal_dir: str
+    workdir: str
+    command: str
+    decision: str
+    output: str
+
+
+def _gate(state: _GateState) -> dict[str, str]:
+    """Interrupt before any later node. Resume re-enters here, then continues."""
+    decision = interrupt({"approval_id": state.get("approval_id", "")})
+    return {"decision": str(decision)}
+
+
+def _effect(state: _GateState) -> dict[str, str]:
+    if str(state.get("decision") or "") != "approved":
+        return {"output": ""}
+    box: dict[str, str] = {}
+
+    def _run() -> None:
+        box["text"] = asyncio.run(_execute_approved(state))
+
+    thread = threading.Thread(target=_run)
+    thread.start()
+    thread.join()
+    return {"output": box.get("text", "")}
+
+
+def _compile(saver: object):
+    graph = StateGraph(_GateState)
+    graph.add_node("gate", _gate)
+    graph.add_node("effect", _effect)
+    graph.add_edge(START, "gate")
+    graph.add_edge("gate", "effect")
+    graph.add_edge("effect", END)
+    return graph.compile(checkpointer=saver)
+
+
+async def _execute_approved(state: _GateState) -> str:
+    from RxyCode.RxyCode1_1_0.execution.tool_journal import (
+        ToolExecutionJournal,
+        remote_checkpoint_id,
+    )
+    from RxyCode.RxyCode1_1_0.execution.tool_orchestrator import ToolOrchestrator
+    from RxyCode.RxyCode1_1_0.tools.bash import bash_tool
+
+    journal = ToolExecutionJournal(state["journal_dir"])
+    orchestrator = ToolOrchestrator()
+    orchestrator.register("bash", bash_tool)
+    token = orchestrator.bind_tool_journal(
+        journal,
+        state["attempt_id"],
+        remote_checkpoint_id(state["run_id"]),
+        state["call_id"],
+    )
+    try:
+        return await orchestrator.execute_tool(
+            "bash",
+            {
+                "command": state.get("command", ""),
+                "description": "approved shell",
+                "workdir": state.get("workdir", ""),
+            },
+            {
+                "safety": {
+                    "enabled": True,
+                    "permission_mode": "confirm_all",
+                    "allowed_write_paths": [state.get("workdir", "")],
+                },
+                "execution": {
+                    "workspace_root": state.get("workdir", ""),
+                    "tool_retry_attempts": 1,
+                },
+            },
+            approval_source="explicit_command",
+            call_id=state.get("call_id"),
+        )
+    finally:
+        orchestrator.reset_tool_journal(token)
+
+
+async def _pause_graph(database_url: str, session_id: str, run_id: str, initial: dict) -> None:
+    async with _open_saver(database_url) as saver:
+        graph = _compile(saver)
+        config = {"configurable": {"thread_id": graph_thread_id(session_id, run_id)}}
+        await graph.ainvoke(initial, config)
+
+
+async def _resume_graph(database_url: str, session_id: str, run_id: str) -> None:
+    async with _open_saver(database_url) as saver:
+        graph = _compile(saver)
+        config = {"configurable": {"thread_id": graph_thread_id(session_id, run_id)}}
+        await graph.ainvoke(Command(resume="approved"), config)
+
+
+def _journal_state(journal_dir: str, attempt_id: str, call_id: str) -> str:
+    from RxyCode.RxyCode1_1_0.execution.tool_journal import (
+        RemoteJournalAdapter,
+        ToolExecutionJournal,
+    )
+
+    if not attempt_id or not call_id:
+        return "not_started"
+    return RemoteJournalAdapter(ToolExecutionJournal(journal_dir)).state(
+        attempt_id=attempt_id,
+        call_id=call_id,
+    )
+

+def pause_for_approval(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    call_id: str,
+    tool: str,
+    permission_scope: str,
+    arguments: dict[str, Any],
+    deadline: datetime,
+    worker_id: str,
+    epoch: int,
+    journal_dir: str,
+    workdir: str,
+    command: str,
+) -> dict[str, Any]:
+    """Ask only when the run is not already fully authorized. Never run the tool here."""
+    loaded = load_run(database_url, run_id)
+    if loaded["permission_profile"] == "full_auto":
+        return {"asked": False, "approval_id": None, "executed": False}
+    attempt_id = f"att_{uuid.uuid4().hex}"
+    requested = request_approval(
+        database_url,
+        tenant_id=tenant_id,
+        run_id=run_id,
+        call_id=call_id,
+        tool=tool,
+        permission_scope=permission_scope,
+        arguments=arguments,
+        deadline=deadline,
+        worker_id=worker_id,
+        epoch=epoch,
+        attempt_id=attempt_id,
+    )
+    asyncio.run(
+        _call_on_selector(
+            lambda: _pause_graph(
+                database_url,
+                loaded["session_id"],
+                run_id,
+                {
+                    "approval_id": requested["approval_id"],
+                    "attempt_id": attempt_id,
+                    "call_id": call_id,
+                    "run_id": run_id,
+                    "journal_dir": journal_dir,
+                    "workdir": workdir,
+                    "command": command,
+                },
+            )
+        )
+    )
+    return {
+        "asked": True,
+        "approval_id": requested["approval_id"],
+        "executed": False,
+    }
+

+def resume_approved(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    approval_id: str,
+    worker_id: str,
+    journal_dir: str,
+    workdir: str,
+    command: str,
+) -> dict[str, Any]:
+    """Continue the interrupted graph. A completed or unknown journal does not run again."""
+    del workdir, command
+    record = get_approval(database_url, tenant_id=tenant_id, approval_id=approval_id)
+    if record is None or record["status"] != "approved":
+        return {"executed": False}
+    before = _journal_state(journal_dir, record["attempt_id"], record["call_id"])
+    if before in {"completed", "unknown"}:
+        return {"executed": False}
+    if claim(database_url, run_id, worker_id) is None:
+        return {"executed": False}
+    loaded = load_run(database_url, run_id)
+    asyncio.run(
+        _call_on_selector(
+            lambda: _resume_graph(database_url, loaded["session_id"], run_id)
+        )
+    )
+    after = _journal_state(journal_dir, record["attempt_id"], record["call_id"])
+    return {"executed": before == "not_started" and after == "completed"}

+        "    workdir=os.environ['C06_WORKDIR'],\n"
+        "    command=os.environ['C06_COMMAND'],\n"
+        ")\n"
+        "print('C06_RESUME ' + json.dumps(result))\n"

+    previous = child.get("PYTHONPATH", "")

+        capture_output=True,
+        text=True,
+        timeout=90,
+        check=False,
+    )
+    assert proc.returncode == 0, proc.stderr or proc.stdout
+    line = next(item for item in proc.stdout.splitlines() if item.startswith("C06_RESUME "))
