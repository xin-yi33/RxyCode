+"""Explicit run scope. The server fills it. The client body does not."""
+
+from __future__ import annotations
+
+import uuid
+from collections.abc import Mapping
+from dataclasses import asdict, dataclass
+
+_IDENTITY_FIELDS = (
+    "tenant_id",
+    "actor_id",
+    "project_id",
+    "session_id",
+    "target_id",
+    "permission_profile",
+    "deadline_at",
+    "trace_id",
+)
+_FORBIDDEN = ("api_key", "device_token", "authorization", "token")
+
+
+@dataclass(frozen=True)
+class RunContext:
+    """One run inside one session. Frozen so callers cannot swap tenant later."""
+
+    tenant_id: str
+    actor_id: str
+    project_id: str
+    session_id: str
+    run_id: str
+    target_id: str
+    permission_profile: str
+    deadline_at: str
+    trace_id: str
+
+    def to_dict(self) -> dict[str, str]:
+        return {key: str(value) for key, value in asdict(self).items()}
+
+    @classmethod
+    def from_authenticated(
+        cls,
+        authenticated: Mapping[str, object],
+        claimed: Mapping[str, object] | None = None,
+    ) -> RunContext:
+        """Build a run from the authenticated identity.
+
+        ``claimed`` is the client body. Its tenant id and secrets are ignored.
+        ``run_id`` is always minted here, never copied from either mapping.
+        """
+        del claimed
+        for key in _FORBIDDEN:
+            if key in authenticated:
+                raise ValueError(f"RunContext refuses {key}")
+        values: dict[str, str] = {}
+        for field in _IDENTITY_FIELDS:
+            raw = authenticated.get(field)
+            if raw is None or str(raw) == "":
+                raise ValueError(f"RunContext missing {field}")
+            values[field] = str(raw)
+        values["run_id"] = uuid.uuid4().hex
+        return cls(**values)

+async def test_w1_c02_integration():
+    """Two concurrent ToolOrchestrator calls do not swap tenant or target."""
+    from langchain_core.tools import StructuredTool
+
+    from appserver.runtime import bind_prompt_context, current_run_context, reset_prompt_context
+    from execution.tool_orchestrator import ToolOrchestrator
+    from rxycode_protocol.run_context import RunContext
+
+    async def echo() -> str:
+        context = current_run_context()
+        assert context is not None
+        return f"{context.tenant_id}|{context.target_id}"
+
+    orchestrator = ToolOrchestrator()
+    orchestrator.register(
+        "echo",
+        StructuredTool.from_function(coroutine=echo, name="echo", description="read run scope"),
+    )
+
+    async def once(tenant: str, target: str) -> str:
+        context = RunContext.from_authenticated(_auth(tenant, target))
+        tokens = bind_prompt_context("ses-c02", None, run_context=context)
+        try:
+            return await orchestrator.execute_tool(
+                "echo",
+                {},
+                config={"safety": {"enabled": False}},
+            )
+        finally:
+            reset_prompt_context(tokens)
+
+    left, right = await asyncio.gather(
+        once("tenant-a", "target-a"),
+        once("tenant-b", "target-b"),
+    )
+    assert left == "tenant-a|target-a"
+    assert right == "tenant-b|target-b"

+"""WEB-W1-C02. One Session opens many runs; the server owns tenant identity."""
+from __future__ import annotations
+
+import sys
+from pathlib import Path
+
+import pytest
+
+REPO = Path(__file__).resolve().parents[3]
+_SRC = REPO / "packages" / "rxycode-protocol" / "src"
+if _SRC.is_dir() and str(_SRC) not in sys.path:
+    sys.path.insert(0, str(_SRC))
+
+AUTH = {
+    "tenant_id": "tenant-real",
+    "actor_id": "actor-1",
+    "project_id": "project-1",
+    "session_id": "client-must-not-win",
+    "target_id": "target-1",
+    "permission_profile": "default",
+    "deadline_at": "2026-10-16T00:00:00Z",
+    "trace_id": "trace-1",
+}
+
+
+async def test_w1_c02_contract(tmp_path: Path):
+    """Two runs on one Session get different run ids and do not rebuild it."""
+    from core.session import Session
+    from rxycode_protocol.run_context import RunContext
+
+    class _Agent:
+        async def run(self, text: str, mode: str = "build") -> str:
+            del text, mode
+            return "ok"
+
+        def cancel(self) -> bool:
+            return True
+
+    emitted: list[object] = []
+    session = Session(session_id="ses-c02", workspace_root=tmp_path, emit=emitted.append)
+    same = session
+    first = session.open_run(AUTH)
+    second = session.open_run(AUTH)
+    assert isinstance(first, RunContext)
+    assert isinstance(second, RunContext)
+    assert first.run_id != second.run_id
+    assert first.session_id == second.session_id == "ses-c02"
+    assert session is same
+    assert session.emit == emitted.append
+    await session.prompt(_Agent(), "turn", mode="build", run_id="legacy-run", run_context=second)
+    assert session is same
+    assert session._active_run is second
+

+def test_w1_c02_negative():
+    """A forged tenant id and secret fields never enter the server context."""
+    from rxycode_protocol.run_context import RunContext
+
+    claimed = {
+        "tenant_id": "forged-tenant",
+        "api_key": "sk-live-secret",
+        "device_token": "device-secret",
+    }
+    context = RunContext.from_authenticated(AUTH, claimed=claimed)
+    blob = str(context.to_dict())
+    assert context.tenant_id == "tenant-real"

+    assert "sk-live-secret" not in blob
+    assert "device-secret" not in blob

+        RunContext.from_authenticated({**AUTH, "api_key": "sk-live-secret"})
