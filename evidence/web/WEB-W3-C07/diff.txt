+import { useEffect, useRef, useState } from "react"
+
+import { createClient } from "../../sdk/client"

+export function RunStatusBanner({
+  runId,
+  onStatus,
+}: {
+  runId: string
+  onStatus?: (status: string) => void
+}) {
+  const openerRef = useRef<HTMLButtonElement>(null)
+  const dialogRef = useRef<HTMLDivElement>(null)
+  const [open, setOpen] = useState(false)
+  const [failed, setFailed] = useState(false)
+
+  useEffect(() => {
+    let cancelled = false
+    const probe = async () => {
+      try {
+        const response = await fetch(`/api/v1/runs/${encodeURIComponent(runId)}`, {
+          credentials: "same-origin",
+          headers: { accept: "application/json" },
+        })
+        if (!cancelled) setFailed(!response.ok)
+      } catch {
+        if (!cancelled) setFailed(true)
+      }
+    }
+    void probe()
+    return () => {
+      cancelled = true
+    }
+  }, [runId])
+
+  useEffect(() => {
+    if (!open) return
+    const dialog = dialogRef.current
+    if (!dialog) return
+    const focusable = () =>
+      [...dialog.querySelectorAll<HTMLElement>("button, a[href], input, select, textarea")]
+    focusable()[0]?.focus()
+    const onKey = (event: KeyboardEvent) => {
+      if (event.key === "Escape") {
+        event.preventDefault()
+        setOpen(false)
+        openerRef.current?.focus()
+        return
+      }
+      if (event.key !== "Tab") return
+      const items = focusable()
+      if (items.length === 0) return
+      const first = items[0]
+      const last = items[items.length - 1]
+      const active = document.activeElement
+      if (!(active instanceof HTMLElement) || !dialog.contains(active)) {
+        event.preventDefault()
+        first.focus()
+        return
+      }
+      if (event.shiftKey && active === first) {
+        event.preventDefault()
+        last.focus()
+      } else if (!event.shiftKey && active === last) {
+        event.preventDefault()
+        first.focus()
+      }
+    }
+    document.addEventListener("keydown", onKey)
+    return () => document.removeEventListener("keydown", onKey)
+  }, [open])
+
+  if (failed) {
+    return (
+      <div role="alert">
+        <p>连接失败</p>
+        <button type="button" onClick={() => window.location.reload()}>
+          重试
+        </button>
+      </div>
+    )
+  }
+
+  return (
+    <div>
+      <p>运行状态以文字显示，不单靠颜色。</p>
+      <button ref={openerRef} type="button" onClick={() => setOpen(true)}>
+        打开审批
+      </button>
+      <button
+        type="button"
+        onClick={() => {
+          void createClient()
+            .cancelRun(runId)
+            .then((result) => onStatus?.(result.status))
+        }}
+      >
+        取消
+      </button>
+      {open ? (
+        <div ref={dialogRef} role="dialog" aria-modal="true" aria-label="审批">
+          <button type="button" onClick={() => { setOpen(false); openerRef.current?.focus() }}>
+            关闭
+          </button>
+          <button type="button">返回</button>
+        </div>
+      ) : null}
+    </div>
+  )

+            ).scalar_one_or_none()
+            if row is None:
+                raise HTTPException(status_code=404, detail="RUN_NOT_FOUND")
+            if row.status in _DONE:
+                raise HTTPException(status_code=409, detail="RUN_TERMINAL")

+                seq = max(int(row.event_seq or 0), int(retained or 0)) + 1
+                row.event_seq = seq
+                event_id = uuid.uuid4().hex
+                db.add(
+                    RunEvent(
+                        event_id=event_id,
+                        run_id=row.id,
+                        tenant_id=record.tenant_id,
+                        seq=seq,
+                        type="run.status",
+                        payload={"status": "cancel_requested"},
+                        at=datetime.now(timezone.utc),
+                    )
+                )

+                        attempts=0,
+                    )
+                )

+def test_w3_c07_contract() -> None:
+    """Opening the approval dialog traps focus and restores it on close."""
+    assert BANNER.is_file(), "RunStatusBanner.tsx is missing"
+    assert RESPONSIVE.is_file(), "responsive.css is missing"
+    stack = _c02._Stack()
+    try:
+        with sync_playwright() as playwright:
+            browser = playwright.chromium.launch()
+            page = browser.new_page()
+            _c02._login(page, stack.origin)
+            run_id, _approval_id = _c06._seed(_c06.datetime(2099, 1, 2, tzinfo=_c06.timezone.utc))
+            page.goto(f"{stack.origin}/runs/{run_id}", wait_until="domcontentloaded")
+            opener = page.get_by_role("button", name="打开审批")
+            opener.focus()
+            opener.press("Enter")
+            dialog = page.get_by_role("dialog")
+            expect(dialog).to_be_visible()
+            page.keyboard.press("Tab")
+            assert dialog.locator(":focus").count() == 1
+            page.keyboard.press("Escape")
+            expect(opener).to_be_focused()
+            browser.close()
+    finally:
+        stack.close()
+

+def test_w3_c07_negative() -> None:
+    """An unreachable API shows a recoverable error instead of a blank page."""
+    assert BANNER.is_file(), "RunStatusBanner.tsx is missing"
+    stack = _c02._Stack()

+        with sync_playwright() as playwright:

+            assert page.locator("body").inner_text().strip() != ""
+            browser.close()
+    finally:
+        stack.close()
