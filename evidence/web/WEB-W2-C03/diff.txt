+"""Create a durable Run, its first event, and the outbox row in one transaction."""
+
+from __future__ import annotations
+
+import hashlib
+import json
+import uuid
+from dataclasses import dataclass
+from datetime import datetime, timezone
+
+from sqlalchemy import create_engine, select
+from sqlalchemy.exc import IntegrityError
+from sqlalchemy.orm import Session
+from sqlalchemy.pool import NullPool
+
+from rxycode_server.persistence.models import Outbox, Run, RunEvent
+
+
+class RunConflict(Exception):
+    """The idempotency key already belongs to a different payload."""
+
+
+class RunAlreadyActive(Exception):
+    """This session already has an active Run."""
+
+
+@dataclass(frozen=True)
+class RunAccepted:
+    http_status: int
+    run_id: str
+    status: str
+    events_url: str
+
+
+def _digest(message: str, target_id: str, permission_profile: str, request_id: str) -> str:
+    payload = json.dumps(
+        {
+            "message": message,
+            "permission_profile": permission_profile,
+            "request_id": request_id,
+            "target_id": target_id,
+        },
+        sort_keys=True,
+        separators=(",", ":"),
+    ).encode()
+    return hashlib.sha256(payload).hexdigest()
+
+
+def _accepted(run_id: str) -> RunAccepted:
+    return RunAccepted(
+        http_status=202,
+        run_id=run_id,
+        status="queued",
+        events_url=f"/api/v1/runs/{run_id}/events",
+    )
+
+
+def create_run(
+    database_url: str,
+    *,
+    tenant_id: str,
+    session_id: str,
+    message: str,
+    target_id: str,
+    permission_profile: str,
+    request_id: str,
+    claimed_tenant_id: str | None = None,
+) -> RunAccepted:
+    """Insert run + event + outbox. The caller's tenant wins over any body claim."""
+    del claimed_tenant_id
+    digest = _digest(message, target_id, permission_profile, request_id)
+    engine = create_engine(database_url, poolclass=NullPool)
+    try:
+        with Session(engine) as db:
+            existing = db.scalars(
+                select(Run).where(Run.tenant_id == tenant_id, Run.request_id == request_id)
+            ).one_or_none()
+            if existing is not None:
+                if existing.input_digest != digest:
+                    raise RunConflict(request_id)
+                return _accepted(existing.id)
+            run_id = uuid.uuid4().hex
+            event_id = uuid.uuid4().hex
+            try:
+                db.add(
+                    Run(
+                        id=run_id,
+                        tenant_id=tenant_id,
+                        session_id=session_id,
+                        status="pending",
+                        revision=0,
+                        lease_epoch=0,
+                        request_id=request_id,
+                        input_digest=digest,
+                        target_id=target_id,
+                    )
+                )
+                db.flush()
+                db.add(
+                    RunEvent(
+                        event_id=event_id,
+                        run_id=run_id,
+                        tenant_id=tenant_id,
+                        seq=1,
+                        type="run.created",
+                        payload={"request_id": request_id, "target_id": target_id},
+                        at=datetime.now(timezone.utc),
+                    )
+                )
+                db.flush()
+                db.add(
+                    Outbox(
+                        id=uuid.uuid4().hex,
+                        event_id=event_id,
+                        dispatch_status="pending",
+                        attempts=0,
+                    )
+                )
+                db.commit()
+            except IntegrityError as exc:
+                db.rollback()
+                detail = str(exc.orig)
+                if "uq_runs_one_active_per_session" in detail:
+                    raise RunAlreadyActive(session_id) from exc
+                if "uq_runs_tenant_request" in detail:
+                    winner = db.scalars(
+                        select(Run).where(Run.tenant_id == tenant_id, Run.request_id == request_id)
+                    ).one()
+                    if winner.input_digest != digest:
+                        raise RunConflict(request_id) from exc
+                    return _accepted(winner.id)
+                raise
+    finally:
+        engine.dispose()
+    return _accepted(run_id)

+"""Publish pending outbox rows to a Redis stream. A failed publish stays pending."""
+
+from __future__ import annotations
+
+from sqlalchemy import create_engine, text
+from sqlalchemy.pool import NullPool
+
+
+def dispatch_pending(database_url: str, client, *, stream: str = "rxycode:runs") -> int:
+    """Send each pending outbox row once. A transport error increments attempts only."""
+    engine = create_engine(database_url, poolclass=NullPool)
+    sent = 0
+    try:
+        with engine.begin() as conn:
+            rows = conn.execute(
+                text(
+                    """
+                    SELECT o.id, o.event_id, e.run_id, e.seq, e.type
+                    FROM outbox o
+                    JOIN run_events e ON e.event_id = o.event_id
+                    WHERE o.dispatch_status = 'pending'
+                    ORDER BY e.seq
+                    FOR UPDATE OF o
+                    """
+                )
+            ).all()
+            for row in rows:
+                try:
+                    client.xadd(
+                        stream,
+                        {
+                            "event_id": row.event_id,
+                            "run_id": row.run_id,
+                            "seq": str(row.seq),
+                            "type": row.type,
+                        },
+                    )
+                except Exception:
+                    conn.execute(
+                        text("UPDATE outbox SET attempts = attempts + 1 WHERE id = :id"),
+                        {"id": row.id},
+                    )
+                    continue
+                conn.execute(
+                    text("UPDATE outbox SET dispatch_status = 'sent' WHERE id = :id"),
+                    {"id": row.id},
+                )
+                sent += 1
+    finally:
+        engine.dispose()
+    return sent

+"""Claim a Run with a fenced lease. A stale epoch cannot commit a terminal status."""
+
+from __future__ import annotations
+
+from datetime import datetime, timedelta, timezone
+
+from sqlalchemy import create_engine, text
+from sqlalchemy.pool import NullPool
+
+
+class LeaseRejected(Exception):
+    """The worker's lease epoch is no longer the one stored in the database."""
+
+
+def _engine(database_url: str):
+    return create_engine(database_url, poolclass=NullPool)
+
+
+def claim(database_url: str, run_id: str, worker_id: str, *, ttl_seconds: int = 30) -> int | None:
+    """Take the Run if it is free or its previous lease has expired."""
+    now = datetime.now(timezone.utc)
+    until = now + timedelta(seconds=ttl_seconds)
+    engine = _engine(database_url)
+    try:
+        with engine.begin() as conn:
+            row = conn.execute(
+                text(
+                    """
+                    UPDATE runs
+                    SET status = 'leased',
+                        lease_owner = :worker,
+                        lease_epoch = lease_epoch + 1,
+                        lease_until = :until,
+                        revision = revision + 1
+                    WHERE id = :run_id
+                      AND status IN ('pending', 'leased', 'running')
+                      AND (lease_owner IS NULL OR lease_until < :now)
+                    RETURNING lease_epoch
+                    """
+                ),
+                {"worker": worker_id, "until": until, "now": now, "run_id": run_id},
+            ).first()
+    finally:
+        engine.dispose()
+    if row is None:
+        return None
+    return int(row[0])
+
+
+def commit_terminal(
+    database_url: str,
+    run_id: str,
+    worker_id: str,
+    epoch: int,
+    status: str,
+) -> None:
+    """Write a terminal status only when this epoch still owns an unexpired lease."""
+    if status not in {"succeeded", "failed", "cancelled"}:
+        raise ValueError(f"terminal status required, got {status}")
+    now = datetime.now(timezone.utc)
+    engine = _engine(database_url)
+    try:
+        with engine.begin() as conn:
+            row = conn.execute(
+                text(
+                    """
+                    UPDATE runs
+                    SET status = :status,
+                        revision = revision + 1
+                    WHERE id = :run_id
+                      AND lease_owner = :worker
+                      AND lease_epoch = :epoch
+                      AND lease_until >= :now
+                    RETURNING id
+                    """
+                ),
+                {
+                    "status": status,
+                    "run_id": run_id,
+                    "worker": worker_id,
+                    "epoch": epoch,
+                    "now": now,
+                },
+            ).first()
+    finally:
+        engine.dispose()
+    if row is None:
+        raise LeaseRejected(run_id)
+
+
+def attempt_tool(journal, attempt_id: str, tool: str, args: dict, execute):
+    """Reserve the existing tool journal. A network failure leaves the write pending."""
+    binding = journal.binding(attempt_id)
+    call = binding.next_call(tool, args)
+    reservation = journal.reserve(attempt_id, call)
+    if reservation.action != "execute":
+        return {
+            "executed": False,
+            "action": reservation.action,
+            "run_status": "recovery_required",
+        }
+    try:
+        result = execute()
+    except OSError:
+        return {"executed": True, "action": "uncertain", "run_status": "recovery_required"}
+    journal.complete(attempt_id, call, result)
+    return {"executed": True, "action": "execute", "run_status": "running", "result": result}
