+_SESSION_TTL = 12 * 60 * 60
+_STATE_TTL = 10 * 60
+
+
+@dataclass
+class OAuthState:
+    verifier: str
+    created: float
+
+
+@dataclass
+class AuthRecord:
+    user_id: str

+def db_session(request: Request) -> Iterator[Session]:
+    db = Session(request.app.state.engine)
+    try:
+        yield db
+        db.commit()
+    except Exception:
+        db.rollback()
+        raise
+    finally:
+        db.close()
+
+
+def _hash_token(secret: str, token: str) -> str:
+    return hmac.new(secret.encode(), token.encode(), "sha256").hexdigest()
+
+
+def _secure(request: Request) -> bool:
+    return bool(request.app.state.secure)
+
+
+def require_write(request: Request) -> None:
+    origin = request.headers.get("origin", "")
+    if origin not in request.app.state.allowed_origins:
+        raise HTTPException(status_code=403, detail="FORBIDDEN")
+    cookie = request.cookies.get("rxy_csrf", "")
+    header = request.headers.get("x-csrf-token", "")
+    if not cookie or not header or not hmac.compare_digest(cookie, header):
+        raise HTTPException(status_code=403, detail="FORBIDDEN")
+
+
+def current_record(request: Request) -> AuthRecord:
+    raw = request.cookies.get("rxy_session", "")
+    if not raw:
+        raise HTTPException(status_code=401, detail="UNAUTHORIZED")
+    record = request.app.state.sessions.get(_hash_token(request.app.state.secret, raw))
+    if record is None or time.monotonic() - record.created > _SESSION_TTL:
+        raise HTTPException(status_code=401, detail="UNAUTHORIZED")
+    return record
+
+
+def _drop_current(request: Request) -> None:
+    raw = request.cookies.get("rxy_session", "")
+    if not raw:
+        return
+    request.app.state.sessions.pop(_hash_token(request.app.state.secret, raw), None)
+
+
+def _issue(request: Request, user_id: str, tenant_id: str) -> str:
+    token = secrets.token_urlsafe(32)
+    request.app.state.sessions[_hash_token(request.app.state.secret, token)] = AuthRecord(
+        user_id=user_id,
+        tenant_id=tenant_id,
+        created=time.monotonic(),
+    )
+    return token
+
+
+def _set_session(response: Response, token: str, *, secure: bool) -> None:
+    response.set_cookie(
+        "rxy_session",
+        token,
+        httponly=True,
+        secure=secure,
+        samesite="lax",
+        path="/",
+        max_age=_SESSION_TTL,
+    )
+
+
+def _local_account(db: Session) -> tuple[User, Organization]:
+    user = User(id=uuid.uuid4().hex, github_id=None, kind="local")
+    org = Organization(id=uuid.uuid4().hex, tenant_id=uuid.uuid4().hex)
+    db.add(user)
+    db.add(org)
+    db.add(Membership(id=uuid.uuid4().hex, tenant_id=org.tenant_id, user_id=user.id, role="owner"))
+    db.flush()
+    return user, org

+def _github_account(db: Session, github_id: str) -> tuple[User, str]:
+    user = db.scalars(select(User).where(User.github_id == github_id)).one_or_none()
+    if user is not None:
+        tenant_id = db.scalars(select(Membership.tenant_id).where(Membership.user_id == user.id)).one()
+        return user, tenant_id
+    user = User(id=uuid.uuid4().hex, github_id=github_id, kind="github")
+    org = Organization(id=uuid.uuid4().hex, tenant_id=uuid.uuid4().hex)
+    db.add(user)
+    db.add(org)
+    db.add(Membership(id=uuid.uuid4().hex, tenant_id=org.tenant_id, user_id=user.id, role="owner"))
+    db.flush()
+    return user, org.tenant_id

+def exchange_github_code(request: Request, code: str, verifier: str) -> str:
+    """Trade a PKCE code at GitHub. The access token is not stored or returned."""
+    app = request.app
+    payload = {
+        "client_id": app.state.github_client_id,
+        "client_secret": app.state.github_client_secret,
+        "code": code,
+        "code_verifier": verifier,
+        "redirect_uri": app.state.github_redirect_uri,
+        "grant_type": "authorization_code",
+    }
+    with httpx.Client(transport=app.state.github_transport) as client:
+        token_response = client.post(
+            "https://github.com/login/oauth/access_token",
+            data=payload,
+            headers={"Accept": "application/json"},
+        )
+        token_response.raise_for_status()
+        access_token = str(token_response.json().get("access_token", ""))
+        user_response = client.get(
+            "https://api.github.com/user",
+            headers={"Authorization": f"Bearer {access_token}", "Accept": "application/json"},
+        )
+        user_response.raise_for_status()
+        github_id = str(user_response.json()["id"])
+    return github_id

+    verifier = secrets.token_urlsafe(48)
+    digest = hashlib.sha256(verifier.encode()).digest()
+    challenge = base64.urlsafe_b64encode(digest).rstrip(b"=").decode()
+    state = secrets.token_urlsafe(32)
+    request.app.state.oauth_states[state] = OAuthState(verifier=verifier, created=time.monotonic())
+    query = httpx.QueryParams(
+        {
+            "client_id": request.app.state.github_client_id,
+            "redirect_uri": request.app.state.github_redirect_uri,
+            "state": state,
+            "code_challenge": challenge,
+            "code_challenge_method": "S256",
+            "scope": "read:user",
+        }
+    )
+    return {
+        "state": state,
+        "code_challenge": challenge,
+        "authorization_url": f"https://github.com/login/oauth/authorize?{query}",
+    }

+"""Web FastAPI gateway. The legacy Ink app in api_server stays a loopback fallback."""
+
+from __future__ import annotations
+
+from contextlib import asynccontextmanager
+
+from fastapi import FastAPI, HTTPException, Request
+from fastapi.responses import JSONResponse
+from sqlalchemy import create_engine
+from sqlalchemy.pool import NullPool
+
+from rxycode_server.api.auth import router as auth_router
+from rxycode_server.api.projects import router as projects_router
+
+_PLACEHOLDERS = ("replace-with", "change-me", "changeme", "example-token", "example-secret")
+
+
+def _require_secret(secret: str) -> str:
+    value = (secret or "").strip()
+    folded = value.casefold()
+    if len(value) < 32 or len(set(value)) < 10 or any(marker in folded for marker in _PLACEHOLDERS):
+        raise RuntimeError("web gateway requires a local random secret")
+    return value
+
+
+def create_app(
+    database_url: str,
+    *,
+    mode: str,
+    secret: str,
+    allowed_origins: list[str],
+    github_client_id: str = "",
+    github_client_secret: str = "",
+    github_redirect_uri: str = "",
+    github_transport: object | None = None,
+) -> FastAPI:
+    """Build the cloud or local gateway. It does not mount the legacy /chat app."""
+    checked = _require_secret(secret)
+    if mode not in {"local", "cloud"}:
+        raise RuntimeError("mode must be local or cloud")
+
+    @asynccontextmanager
+    async def lifespan(app: FastAPI):
+        yield
+        app.state.engine.dispose()
+
+    app = FastAPI(title="RxyCode Web", version="1.4.3", lifespan=lifespan)
+    app.state.engine = create_engine(database_url, poolclass=NullPool)
+    app.state.secret = checked
+    app.state.mode = mode
+    app.state.secure = mode == "cloud"
+    app.state.allowed_origins = tuple(allowed_origins)
+    app.state.sessions = {}
+    app.state.oauth_states = {}
+    app.state.github_client_id = github_client_id
+    app.state.github_client_secret = github_client_secret
+    app.state.github_redirect_uri = github_redirect_uri
+    app.state.github_transport = github_transport
+    app.include_router(auth_router)
+    app.include_router(projects_router)
+
+    @app.exception_handler(HTTPException)
+    async def http_error(_request: Request, exc: HTTPException) -> JSONResponse:
+        code = exc.detail if isinstance(exc.detail, str) else "FORBIDDEN"
+        return JSONResponse(status_code=exc.status_code, content={"code": code})
+
+    return app
+
+
+def serve(
+    *,
+    mode: str,
+    host: str,
+    database_url: str,
+    secret: str,
+    allowed_origins: list[str],
+    port: int = 8766,
+    **kwargs: object,
+) -> FastAPI:
+    """Return the web app. Local mode refuses any non-loopback bind of this gateway."""
+    del port
+    if mode == "local":
+        from RxyCode.RxyCode1_1_0.api_server import reject_as_public_web_gateway
+
+        reject_as_public_web_gateway(host)
+    return create_app(
+        database_url,
+        mode=mode,
+        secret=secret,
+        allowed_origins=allowed_origins,
+        **kwargs,
+    )

+def _headers(token: str, origin: str = ORIGIN) -> dict[str, str]:
+    return {"X-CSRF-Token": token, "Origin": origin}
+
+
+def _login_cookie(app, token: str, origin: str) -> str:
+    """Read the session Set-Cookie from the ASGI response itself."""
+    messages: list[dict] = []
+
+    async def receive():
+        return {"type": "http.request", "body": b"", "more_body": False}
+
+    async def send(message):
+        messages.append(message)
+
+    scope = {
+        "type": "http",
+        "asgi": {"version": "3.0"},
+        "http_version": "1.1",
+        "method": "POST",
+        "scheme": "https",
+        "path": "/api/v1/auth/local/login",
+        "raw_path": b"/api/v1/auth/local/login",
+        "query_string": b"",
+        "headers": [
+            (b"origin", origin.encode()),
+            (b"x-csrf-token", token.encode()),
+            (b"cookie", f"rxy_csrf={token}".encode()),
+        ],
+        "client": ("127.0.0.1", 50000),
+        "server": ("testserver", 443),
+    }
+    asyncio.run(app(scope, receive, send))
+    start = next(item for item in messages if item["type"] == "http.response.start")
+    assert start["status"] == 200
+    chunks = [
+        value.decode()
+        for key, value in start["headers"]
+        if key.lower() == b"set-cookie"
+    ]
+    return "\n".join(chunks).casefold()
+

+def _csrf(client) -> str:
+    response = client.get("/api/v1/auth/csrf")
+    assert response.status_code == 200
+    return response.json()["csrf_token"]

+def test_w2_c02_contract(postgres_url: str):
+    """A real login is readable from GET /me, and the legacy app is not the gateway."""
+    upgrade_head(postgres_url)
+    with pytest.raises(RuntimeError, match="public web gateway"):
+        serve(
+            mode="local",
+            host="0.0.0.0",
+            database_url=postgres_url,
+            secret=SECRET,
+            allowed_origins=[ORIGIN],
+        )
+    app = create_app(
+        postgres_url,
+        mode="local",
+        secret=SECRET,
+        allowed_origins=[ORIGIN],
+        github_client_id="iv1.client",
+        github_client_secret="github-secret-not-stored",
+        github_redirect_uri="http://127.0.0.1/api/v1/auth/github/callback",
+    )
+    paths = {getattr(route, "path", None) for route in app.routes}
+    assert "/chat" not in paths
+    assert "/api/v1/me" in app.openapi()["paths"]
+    client = TestClient(app)
+    token = _csrf(client)
+    first = client.post("/api/v1/auth/local/login", headers=_headers(token))
+    assert first.status_code == 200
+    old_cookie = client.cookies.get("rxy_session")
+    me = client.get("/api/v1/me")
+    assert me.status_code == 200
+    body = me.json()
+    engine = create_engine(postgres_url)
+    with Session(engine) as db:
+        user = db.get(User, body["user"]["id"])
+        org = db.scalars(
+            select(Organization).where(Organization.tenant_id == body["organization"]["tenant_id"])
+        ).one()
+    assert user is not None
+    assert body["user"]["kind"] == user.kind == "local"
+    assert body["organization"]["tenant_id"] == org.tenant_id
+    assert body["membership"]["role"] == "owner"
+    token = _csrf(client)
+    again = client.post("/api/v1/auth/local/login", headers=_headers(token))
+    assert again.status_code == 200
+    stale = TestClient(app)
+    stale.cookies.set("rxy_session", old_cookie)
+    denied = stale.get("/api/v1/me")
+    assert denied.status_code == 401
+    assert denied.json()["code"] == "UNAUTHORIZED"
+    started = client.get("/api/v1/auth/github/start")
+    assert started.status_code == 200
+    state = started.json()["state"]
+    challenge = started.json()["code_challenge"]
+    verifier = app.state.oauth_states[state].verifier
+    digest = hashlib.sha256(verifier.encode()).digest()
+    expected = base64.urlsafe_b64encode(digest).rstrip(b"=").decode()
+    assert challenge == expected
+    assert "code_challenge=" + challenge in started.json()["authorization_url"]
+
+    def handler(request: httpx.Request) -> httpx.Response:
+        if request.url.host == "github.com":
+            sent = request.content.decode()
+            assert f"code_verifier={verifier}" in sent
+            assert "client_secret=github-secret-not-stored" in sent
+            return httpx.Response(200, json={"access_token": "ghs_secret", "token_type": "bearer"})
+        assert request.headers["Authorization"] == "Bearer ghs_secret"
+        return httpx.Response(200, json={"id": 4242, "login": "octo"})
+
+    app.state.github_transport = httpx.MockTransport(handler)
+    callback = client.get("/api/v1/auth/github/callback", params={"state": state, "code": "abc"})
+    assert callback.status_code == 200
+    github_me = client.get("/api/v1/me")
+    assert github_me.status_code == 200
+    assert github_me.json()["user"]["github_id"] == "4242"
+    assert "ghs_secret" not in github_me.text
+    with Session(engine) as db:
+        stored = db.scalars(select(User).where(User.github_id == "4242")).one()
+    assert stored.kind == "github"
+    cloud = create_app(
+        postgres_url,
+        mode="cloud",
+        secret=SECRET,
+        allowed_origins=[CLOUD],
+    )
+    cloud_client = TestClient(cloud)
+    cloud_token = _csrf(cloud_client)
+    cookie = _login_cookie(cloud, cloud_token, CLOUD)
+    assert "httponly" in cookie
+    assert "samesite=lax" in cookie
+    assert "secure" in cookie

+def test_w2_c02_negative(postgres_url: str):
+    """A missing CSRF token or a foreign Origin does not insert a project."""
+    upgrade_head(postgres_url)
+    app = create_app(
+        postgres_url,
+        mode="local",
+        secret=SECRET,
+        allowed_origins=[ORIGIN],
+    )
+    engine = create_engine(postgres_url)
+    client = TestClient(app)
+    token = _csrf(client)
+    assert client.post("/api/v1/auth/local/login", headers=_headers(token)).status_code == 200
+    token = _csrf(client)
+    missing = client.post(
+        "/api/v1/projects",
+        json={"name": "nope", "repo_config": {}},
+        headers={"Origin": ORIGIN},
+    )
+    evil = client.post(
+        "/api/v1/projects",
+        json={"name": "evil", "repo_config": {}},
+        headers=_headers(token, "https://evil.example"),
+    )
+    leaked = client.post(
+        "/api/v1/projects",
+        json={"name": "keyed", "repo_config": {"api_key": "sk-live"}},
+        headers=_headers(token),
+    )
+    assert missing.status_code == 403
+    assert evil.status_code == 403
+    assert leaked.status_code == 400
+    with Session(engine) as db:
+        count = db.scalar(select(func.count()).select_from(Project))
+    assert count == 0
