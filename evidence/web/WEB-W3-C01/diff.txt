+function Shell({ children }: { children: ReactNode }) {
+  const theme = useUi((state) => state.theme)
+  const setTheme = useUi((state) => state.setTheme)
+  const setOffline = useUi((state) => state.setOffline)
+  useEffect(() => {
+    document.documentElement.classList.toggle("dark", theme === "dark")
+    document.documentElement.dataset.theme = theme
+  }, [theme])
+  useEffect(() => {
+    const sync = () => setOffline(!navigator.onLine)
+    sync()
+    window.addEventListener("online", sync)
+    window.addEventListener("offline", sync)
+    return () => {
+      window.removeEventListener("online", sync)
+      window.removeEventListener("offline", sync)
+    }
+  }, [setOffline])
+  return (
+    <div className="min-h-screen bg-background text-foreground">
+      <header className="flex items-center justify-between border-b border-border px-4 py-3">
+        <p className="text-sm font-semibold">RxyCode</p>
+        <Button
+          variant="outline"
+          onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
+        >
+          {theme === "dark" ? "浅色" : "深色"}
+        </Button>
+      </header>
+      <main className="mx-auto max-w-3xl px-4 py-8">{children}</main>
+    </div>
+  )

+export type Me = { user_id: string; tenant_id: string }
+
+export type Project = { id: string; name: string }
+
+export class ApiError extends Error {
+  readonly status: number
+  readonly code: string
+
+  constructor(status: number, code: string) {
+    super(code)
+    this.name = "ApiError"
+    this.status = status
+    this.code = code
+  }
+}
+
+async function readJson(path: string, signal?: AbortSignal): Promise<unknown> {
+  const response = await fetch(path, {
+    signal,
+    headers: { accept: "application/json" },
+  })
+  if (!response.ok) {
+    let code = "UNAVAILABLE"
+    try {
+      const body = (await response.json()) as { code?: string }
+      if (body.code) code = body.code
+    } catch {
+      code = "UNAVAILABLE"
+    }
+    throw new ApiError(response.status, code)
+  }
+  return response.json()
+}
+
+export function createClient() {
+  return {
+    getMe(signal?: AbortSignal): Promise<Me> {
+      return readJson("/api/v1/me", signal) as Promise<Me>
+    },
+    listProjects(signal?: AbortSignal): Promise<Project[]> {
+      return readJson("/api/v1/projects", signal).then((body) => {
+        const record = body as { projects?: Project[] }
+        return record.projects ?? []
+      })
+    },
+    openRunEvents(runId: string): EventSource {
+      return new EventSource(`/api/v1/runs/${encodeURIComponent(runId)}/events`)
+    },
+  }

+_FORBIDDEN = (
+    "electron",
+    "contextBridge",
+    "ipcRenderer",
+    "child_process",

+    "node:child_process",
+    "preload.js",

+def _bun() -> str:
+    found = shutil.which("bun")
+    if found:
+        return found
+    if _BUN.exists():
+        return str(_BUN)
+    raise FileNotFoundError("bun")
+
+
+def _source_digest() -> str:
+    digest = hashlib.sha256()
+    for path in sorted(WEB.rglob("*")):
+        if not path.is_file() or "node_modules" in path.parts or "dist" in path.parts:
+            continue
+        digest.update(str(path.relative_to(WEB)).encode())
+        digest.update(path.read_bytes())
+    return digest.hexdigest()
+
+
+def build_web() -> Path:
+    """Install and build the shipped Vite app. A second call reuses that dist."""
+    package = WEB / "package.json"
+    assert package.is_file(), "apps/web/package.json is missing"
+    dist = WEB / "dist"
+    stamp = dist / ".built"
+    digest = _source_digest()
+    fresh = stamp.is_file() and stamp.read_text(encoding="utf-8") == digest
+    if fresh and (dist / "index.html").is_file():
+        return dist
+    if dist.exists():
+        shutil.rmtree(dist)
+    subprocess.run([_bun(), "install"], cwd=WEB, check=True)
+    subprocess.run([_bun(), "run", "build"], cwd=WEB, check=True)
+    assert (dist / "index.html").is_file()
+    stamp.write_text(digest, encoding="utf-8")
+    return dist
+
+
+def bundle_text(dist: Path) -> str:
+    parts: list[str] = []
+    for path in dist.rglob("*"):
+        if path.is_file() and path.suffix in {".js", ".css", ".html"} and path.name != ".built":
+            parts.append(path.read_text(encoding="utf-8", errors="replace"))
+    return "\n".join(parts)
+
+
+def _serve(dist: Path, hits: list[str]) -> tuple[ThreadingHTTPServer, int]:
+    class Handler(BaseHTTPRequestHandler):
+        def log_message(self, fmt: str, *args: object) -> None:
+            del fmt, args
+
+        def do_GET(self) -> None:  # noqa: N802
+            path = urlparse(self.path).path
+            hits.append(path)
+            if path.startswith("/api/"):
+                body = b'{"code":"UNAVAILABLE"}'
+                self.send_response(503)
+                self.send_header("content-type", "application/json")
+                self.send_header("content-length", str(len(body)))
+                self.end_headers()
+                self.wfile.write(body)
+                return
+            rel = path.lstrip("/") or "index.html"
+            target = dist / rel
+            if not target.is_file():
+                target = dist / "index.html"
+            data = target.read_bytes()
+            self.send_response(200)
+            if target.suffix == ".js":
+                kind = "text/javascript"
+            elif target.suffix == ".css":
+                kind = "text/css"
+            else:
+                kind = "text/html"
+            self.send_header("content-type", kind)
+            self.send_header("content-length", str(len(data)))
+            self.end_headers()
+            self.wfile.write(data)
+
+    server = ThreadingHTTPServer(("127.0.0.1", 0), Handler)
+    thread = threading.Thread(target=server.serve_forever, daemon=True)
+    thread.start()
+    return server, int(server.server_address[1])
+

+def _me_hits(hits: list[str]) -> int:
