+from __future__ import annotations

+import asyncio
+import os
+import subprocess
+import threading
+import uuid
+from pathlib import Path
+from typing import Any

+_GUARD = threading.Lock()
+_SESSIONS: dict[str, "Session"] = {}
+_SCROLL_LIMIT = 200_000
+
+
+def live_pids() -> list[int]:
+    with _GUARD:
+        return [item.pid for item in _SESSIONS.values() if item.exit_code is None]
+
+
+def get_session(session_id: str) -> "Session | None":
+    with _GUARD:
+        return _SESSIONS.get(session_id)
+

+class Session:
+    def __init__(self, argv: list[str], cols: int, rows: int, tenant_id: str, sandboxed: bool) -> None:
+        self.id = uuid.uuid4().hex
+        self.tenant_id = tenant_id
+        self.exit_code: int | None = None
+        self._scroll = ""
+        self._listeners: list[tuple[asyncio.AbstractEventLoop, asyncio.Queue[dict[str, Any]]]] = []
+        self._lock = threading.Lock()
+        self._backend = _spawn(argv, cols, rows, sandboxed)
+        self.pid = self._backend.pid
+        self._backend.start(self._on_output, self._on_exit)
+
+    @property
+    def alive(self) -> bool:
+        return self.exit_code is None
+
+    def text(self) -> str:
+        with self._lock:
+            return self._scroll
+
+    def write(self, data: str) -> None:
+        self._backend.write(data)
+
+    def resize(self, cols: int, rows: int) -> None:
+        self._backend.resize(cols, rows)
+
+    def subscribe(self) -> asyncio.Queue[dict[str, Any]]:
+        loop = asyncio.get_running_loop()
+        queue: asyncio.Queue[dict[str, Any]] = asyncio.Queue()
+        with self._lock:
+            self._listeners.append((loop, queue))
+        return queue
+
+    def unsubscribe(self, queue: asyncio.Queue[dict[str, Any]]) -> None:
+        with self._lock:
+            self._listeners = [item for item in self._listeners if item[1] is not queue]
+
+    def _on_output(self, text: str) -> None:
+        if not text:
+            return
+        with self._lock:
+            self._scroll = (self._scroll + text)[-_SCROLL_LIMIT:]
+            listeners = list(self._listeners)
+        frame = {"type": "output", "data": text}
+        for loop, queue in listeners:
+            loop.call_soon_threadsafe(queue.put_nowait, frame)
+
+    def _on_exit(self, code: int) -> None:
+        with self._lock:
+            self.exit_code = code
+            listeners = list(self._listeners)
+        frame = {"type": "exit", "exit_code": code}
+        for loop, queue in listeners:
+            loop.call_soon_threadsafe(queue.put_nowait, frame)
+

+    def resize(self, cols: int, rows: int) -> None:

+    def terminate(self) -> None:

+        threading.Thread(target=self._wait_loop, args=(on_exit,), daemon=True).start()

+    def write(self, data: str) -> None:
+        os.write(self._master, data.encode("utf-8"))
+
+    def resize(self, cols: int, rows: int) -> None:

+        import termios
+
+        fcntl.ioctl(self._master, termios.TIOCSWINSZ, _winsize(rows, cols))

+    def terminate(self) -> None:
+        self._proc.kill()

+    def _read_loop(self, on_output) -> None:
+        while True:

+def _record(websocket: WebSocket):
+    raw = websocket.cookies.get("rxy_session", "")
+    if not raw:
+        return None
+    record = websocket.app.state.sessions.get(_hash_token(websocket.app.state.secret, raw))
+    if record is None or time.monotonic() - record.created > _SESSION_TTL:

+@router.post("/api/v1/terminals", status_code=201)
+def open_terminal(payload: OpenIn, request: Request) -> dict[str, object]:
+    require_write(request)
+    record = current_record(request)
+    if payload.cols < 1 or payload.rows < 1 or payload.cols > 500 or payload.rows > 500:
+        raise HTTPException(status_code=400, detail="INVALID_INPUT")

+        raise HTTPException(status_code=400, detail="INVALID_INPUT")
+    backend = _runner()

+        session = backend.open_session(
+            list(payload.argv),
+            cols=payload.cols,
+            rows=payload.rows,
+            tenant_id=record.tenant_id,
+            sandboxed=request.app.state.mode == "cloud",
+        )
+    except PermissionError as exc:
+        raise HTTPException(status_code=403, detail="SANDBOX_REQUIRED") from exc
+    return {"id": session.id, "pid": session.pid, "exit_code": None}

+@router.get("/api/v1/terminals/{terminal_id}")
+def read_terminal(terminal_id: str, request: Request) -> dict[str, object]:
+    record = current_record(request)

+    if session is None or session.tenant_id != record.tenant_id:
+        raise HTTPException(status_code=404, detail="NOT_FOUND")
+    return {
+        "id": session.id,
+        "pid": session.pid,
+        "alive": session.alive,
+        "exit_code": session.exit_code,
+        "scrollback": session.text(),
+    }

+@router.websocket("/api/v1/terminals/{terminal_id}/stream")
+async def stream_terminal(websocket: WebSocket, terminal_id: str) -> None:
+    record = _record(websocket)
+    session = _runner().get_session(terminal_id)
+    if record is None or session is None or session.tenant_id != record.tenant_id:
+        await websocket.close(code=1008)
+        return
+    await websocket.accept()
+    queue = session.subscribe()
+    queued = session.text()
+    if queued:
+        await websocket.send_json({"type": "output", "data": queued})
+
+    async def pump() -> None:
+        while True:
+            frame = await queue.get()
+            await websocket.send_json(frame)
+
+    async def incoming() -> None:
+        while True:
+            raw = await websocket.receive_text()
+            message = json.loads(raw)
+            kind = str(message.get("type", ""))
+            if kind == "input":
+                session.write(str(message.get("data", "")))
+            elif kind == "resize":
+                session.resize(int(message["cols"]), int(message["rows"]))
+
+    pump_task = asyncio.create_task(pump())
+    incoming_task = asyncio.create_task(incoming())
+    try:
+        done, pending = await asyncio.wait(
+            {pump_task, incoming_task},
+            return_when=asyncio.FIRST_COMPLETED,
+        )
+        for task in pending:
+            task.cancel()
+        for task in done:
+            task.result()
+    finally:
+        session.unsubscribe(queue)
