+"""Install-time identity for the one historical Session class.
+
+This package does not copy AgentV2. Session and reuse_or_create_session are
+the class and function objects defined in RxyCode.RxyCode1_1_0.core.session.
+"""
+
+from __future__ import annotations
+
+from RxyCode.RxyCode1_1_0.core.session import Session, reuse_or_create_session
+
+__all__ = ["Session", "audit_class_identity", "reuse_or_create_session"]
+
+
+def audit_class_identity(classes: list[type]) -> str:
+    """Return OK when every entry is the same class object, otherwise BLOCKED."""
+    if len(set(classes)) > 1:
+        return "BLOCKED"
+    return "OK"

+from __future__ import annotations

+

+_SRC = REPO / "packages" / "rxycode-core" / "src"
+if _SRC.is_dir() and str(_SRC) not in sys.path:
+    sys.path.insert(0, str(_SRC))

+def test_w1_c01_contract():
+    """Legacy, canonical, and rxycode_core Session are one object."""
+    from RxyCode.RxyCode1_1_0.core.session import Session as Canonical
+    from core.session import Session as Legacy
+    from rxycode_core import Session as Packaged
+
+    assert Legacy is Canonical
+    assert Packaged is Canonical
+
+
+def test_w1_c01_negative():
+    """Two different Agent classes block the import audit."""
+    from RxyCode.RxyCode1_1_0.core.agent_v2 import AgentV2 as Canonical
+    from core.agent_v2 import AgentV2 as Legacy
+    from rxycode_core import audit_class_identity
+
+    assert Legacy is Canonical

+

+"""WEB-W1-C01. An installed wheel exposes one Session and the CLI entry."""
+from __future__ import annotations
+
+import os
+import subprocess
+import sys
+import venv
+from pathlib import Path
+
+REPO = Path(__file__).resolve().parents[3]
+
+
+def _python(env_dir: Path) -> Path:
+    if os.name == "nt":
+        return env_dir / "Scripts" / "python.exe"
+    return env_dir / "bin" / "python"
+
+
+def test_w1_c01_integration(tmp_path: Path):
+    """A fresh venv install keeps Session identity and can start the console."""
+    wheel_dir = tmp_path / "wheel"
+    wheel_dir.mkdir()
+    build = subprocess.run(
+        [
+            sys.executable,
+            "-m",
+            "pip",
+            "wheel",
+            "--no-deps",
+            "-w",
+            str(wheel_dir),
+            str(REPO),
+        ],
+        cwd=REPO,
+        text=True,
+        capture_output=True,
+        check=False,
+    )
+    assert build.returncode == 0, build.stderr
+    wheels = list(wheel_dir.glob("rxycode-*.whl"))
+    assert len(wheels) == 1
+    env_dir = tmp_path / "venv"
+    venv.create(env_dir, with_pip=True, system_site_packages=True)
+    python = _python(env_dir)
+    install = subprocess.run(
+        [str(python), "-m", "pip", "install", "--no-deps", str(wheels[0])],
+        text=True,
+        capture_output=True,
+        check=False,
+    )
+    assert install.returncode == 0, install.stderr
+    probe = subprocess.run(
+        [
+            str(python),
+            "-c",
+            "import RxyCode.RxyCode1_1_0.appserver.server as server\n"
+            "import RxyCode.RxyCode1_1_0.core.session as legacy\n"
+            "import rxycode_core\n"
+            "assert rxycode_core.Session is legacy.Session\n"
+            "print('wheel-session-ok')\n",
+        ],
+        text=True,
+        capture_output=True,
+        check=False,
+    )
+    assert probe.returncode == 0, probe.stderr
+    assert "wheel-session-ok" in probe.stdout
+    console = env_dir / "Scripts" / "rxycode.exe" if os.name == "nt" else env_dir / "bin" / "rxycode"
+    help_run = subprocess.run(
+        [str(console), "--help"],
+        text=True,
+        capture_output=True,
+        check=False,
+    )
+    assert help_run.returncode == 0, help_run.stderr
+    assert "Usage:" in help_run.stdout + help_run.stderr
