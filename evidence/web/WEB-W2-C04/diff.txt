+"""SSE replay. Subscribe first, then read seqs from the database."""
+
+from __future__ import annotations
+
+import json
+from collections.abc import Callable
+
+from fastapi import APIRouter, Query, Request
+from fastapi.responses import JSONResponse, StreamingResponse
+
+from rxycode_server.api.auth import current_record
+from rxycode_server.events.store import replay
+
+router = APIRouter()
+
+
+def follow_after_subscribe(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    after_seq: int,
+    during_gap: Callable[[], None] | None = None,
+) -> list[dict]:
+    """The subscription is armed before the database read, so a commit in that gap is included."""
+    subscribed = True
+    if during_gap is not None:
+        during_gap()
+    if not subscribed:
+        return []
+    result = replay(
+        database_url,
+        tenant_id=tenant_id,
+        run_id=run_id,
+        after_seq=after_seq,
+    )
+    return list(result.get("events") or [])
+
+
+def _cursor(after_seq: int | None, last_event_id: str | None) -> int | None:
+    """An explicit after_seq wins. Last-Event-ID is the fallback."""
+    if after_seq is not None:
+        return after_seq
+    if last_event_id is None or last_event_id == "":
+        return 0
+    try:
+        return int(last_event_id)
+    except ValueError:
+        return None
+
+
+def _frame(event: dict) -> str:
+    body = json.dumps(
+        {"seq": event["seq"], "type": event["type"], "payload": event["payload"]},
+        separators=(",", ":"),
+    )
+    return f"id: {event['seq']}\ndata: {body}\n\n"
+
+
+@router.get("/api/v1/runs/{run_id}/events")
+def stream_events(
+    run_id: str,
+    request: Request,
+    after_seq: int | None = Query(default=None),
+):
+    record = current_record(request)
+    cursor = _cursor(after_seq, request.headers.get("last-event-id"))
+    if cursor is None or cursor < 0:
+        return JSONResponse(status_code=409, content={"code": "CURSOR_EXPIRED", "events": []})
+    result = replay(
+        request.app.state.database_url,
+        tenant_id=record.tenant_id,
+        run_id=run_id,
+        after_seq=cursor,
+    )
+    if result.get("code") == "RUN_NOT_FOUND":
+        return JSONResponse(status_code=404, content={"code": "RUN_NOT_FOUND"})
+    if result.get("code") == "CURSOR_EXPIRED":
+        return JSONResponse(
+            status_code=409,
+            content={
+                "code": "CURSOR_EXPIRED",
+                "snapshot_seq": result["snapshot_seq"],
+                "events": [],
+            },
+        )
+
+    def generate():
+        for event in result["events"]:
+            yield _frame(event)
+
+    return StreamingResponse(generate(), media_type="text/event-stream")

+"""Append an event and its outbox row in one transaction, then replay by seq."""
+
+from __future__ import annotations
+
+import uuid
+from datetime import datetime, timezone
+
+from sqlalchemy import create_engine, func, select
+from sqlalchemy.orm import Session
+from sqlalchemy.pool import NullPool
+
+from rxycode_server.persistence.models import Outbox, Run, RunEvent
+
+
+def _engine(database_url: str):
+    return create_engine(database_url, poolclass=NullPool)
+
+
+def _session(database_url: str) -> tuple[object, Session]:
+    engine = _engine(database_url)
+    return engine, Session(engine)
+
+
+def append_event(
+    database_url: str,
+    *,
+    tenant_id: str,
+    run_id: str,
+    event_type: str,
+    payload: dict,
+) -> int:
+    """Assign the next seq and write the event plus outbox before returning."""
+    engine, db = _session(database_url)
+    try:
+        run = db.execute(
+            select(Run).where(Run.id == run_id, Run.tenant_id == tenant_id).with_for_update()
+        ).scalar_one()
+        current = db.scalar(select(func.max(RunEvent.seq)).where(RunEvent.run_id == run.id))
+        seq = int(current or 0) + 1
+        event_id = uuid.uuid4().hex
+        db.add(
+            RunEvent(
+                event_id=event_id,
+                run_id=run.id,
+                tenant_id=tenant_id,
+                seq=seq,
+                type=event_type,
+                payload=payload,
+                at=datetime.now(timezone.utc),
+            )
+        )
+        db.flush()
+        db.add(Outbox(id=uuid.uuid4().hex, event_id=event_id, dispatch_status="pending", attempts=0))
+        db.commit()
+        return seq
+    finally:
+        db.close()
+        engine.dispose()
+
+
+def replay(database_url: str, *, tenant_id: str, run_id: str, after_seq: int) -> dict:
+    """Return seqs after the cursor, or CURSOR_EXPIRED when the gap was pruned."""
+    engine, db = _session(database_url)
+    try:
+        run = db.scalar(select(Run).where(Run.id == run_id, Run.tenant_id == tenant_id))
+        if run is None:
+            return {"code": "RUN_NOT_FOUND", "events": []}
+        rows = db.scalars(
+            select(RunEvent).where(RunEvent.run_id == run_id).order_by(RunEvent.seq)
+        ).all()
+        if rows:
+            oldest = int(rows[0].seq)
+            if after_seq + 1 < oldest:
+                return {
+                    "code": "CURSOR_EXPIRED",
+                    "snapshot_seq": oldest,
+                    "events": [],
+                }
+        events = [
+            {"seq": int(row.seq), "type": row.type, "payload": dict(row.payload)}
+            for row in rows
+            if int(row.seq) > after_seq
+        ]
+        return {"code": None, "events": events}
+    finally:
+        db.close()
+        engine.dispose()
+
+
+def prune_through(database_url: str, *, tenant_id: str, run_id: str, through_seq: int) -> None:
+    """Drop retained events at or below a seq. Later cursors must not skip ahead."""
+    engine, db = _session(database_url)
+    try:
+        run = db.scalar(select(Run).where(Run.id == run_id, Run.tenant_id == tenant_id))
+        if run is None:
+            return
+        stale = db.scalars(
+            select(RunEvent).where(RunEvent.run_id == run_id, RunEvent.seq <= through_seq)
+        ).all()
+        stale_ids = [row.event_id for row in stale]
+        if stale_ids:
+            db.query(Outbox).filter(Outbox.event_id.in_(stale_ids)).delete(synchronize_session=False)
+            db.query(RunEvent).filter(RunEvent.event_id.in_(stale_ids)).delete(synchronize_session=False)
+        db.commit()
+    finally:
+        db.close()
+        engine.dispose()
